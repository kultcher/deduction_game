## Joint role completions and a persistent Red Herring, without full permutations.
extends RefCounted
const C = preload("res://sim/content.gd")
const Activity = preload("res://workshop/live_activity.gd")
static var cache: Dictionary = {}
static var reuse = true

static func fits(view: Dictionary, world: Dictionary, cards: Array, knowledge: Dictionary, answers: Dictionary = {}) -> bool:
	if not view.get("world_options", {}).get("roles", false):
		return true
	var ids: Array = view.people.map(func(p): return p.id)
	if not reuse:
		return _fits_view(view, ids, world, cards, knowledge, answers)
	var input = [view.people, view.get("timeline", {}), view.get("live_activity", false), cards, knowledge.get("self"), knowledge.get("own_role"), knowledge.get("own_outings", {}), answers, knowledge.get("public_roles", {})]
	var key = hash(input)
	if not cache.has(key) or cache[key].input != input:
		if cache.size() >= 64:
			cache.clear()
		cache[key] = {"input": input.duplicate(true), "results": {}, "seer": seer_flag(cards)}
	var mask: int = coarse_mask(ids.size(), world, cache[key].seer, relay_mediums(ids, cards), sleepwalk_claimants(ids, cards))
	if not cache[key].results.has(mask):
		cache[key].results[mask] = _fits_view(view, ids, world, cards, knowledge, answers)
	return cache[key].results[mask]

## fits() for every world of a universe, hashing the view and cards once rather than
## per world. Same memo as fits(), so the results agree.
## Weight of one role declaration (claim, or a reading's implied gift) by a source who
## is evil in that world: a bluff is possible, but naming this exact role is a choice
## among several, so a claim still weakly supports its claimant (live play only).
const BLUFF_WEIGHT = 0.35

## bluffs (live play): evil sources' role declarations constrain nothing; see _fits.
static func fits_table(view: Dictionary, universe: Array, cards: Array, knowledge: Dictionary, answers: Dictionary = {}, bluffs: bool = false) -> PackedByteArray:
	var out = PackedByteArray()
	out.resize(universe.size())
	if not view.get("world_options", {}).get("roles", false):
		out.fill(1)
		return out
	var ids: Array = view.people.map(func(p): return p.id)
	var input = [view.people, view.get("timeline", {}), view.get("live_activity", false), cards, knowledge.get("self"), knowledge.get("own_role"), knowledge.get("own_outings", {}), answers, knowledge.get("public_roles", {}), bluffs]
	var key = hash(input)
	if not cache.has(key) or cache[key].input != input:
		if cache.size() >= 64:
			cache.clear()
		cache[key] = {"input": input.duplicate(true), "results": {}, "seer": seer_flag(cards)}
	var entry: Dictionary = cache[key]
	var results: Dictionary = entry.results
	var seer: bool = entry.seer
	var mediums: int = relay_mediums(ids, cards)
	var walkers: int = sleepwalk_claimants(ids, cards)
	var routes = null
	for i in range(universe.size()):
		var world: Dictionary = universe[i]
		var mask: int = coarse_mask(ids.size(), world, seer, mediums, walkers)
		if not results.has(mask):
			# The Medium claims a relay adds are the same in every world that passes the
			# relay checks (worlds that fail never reach routing), so routes hoist too.
			if routes == null and Activity.available(view):
				routes = Activity.routes(view, _routing_cards(_live_cards(cards)), answers, knowledge)
			results[mask] = _fits_view(view, ids, world, cards, knowledge, answers, routes, bluffs)
		out[i] = 1 if results[mask] else 0
	return out

## The parts of a world the role check reads, as one int: fits() gives the same answer
## for worlds with the same mask (for the same view, cards and knowledge).
## mediums: relay_mediums(). A relay only reads whether its Medium is Delusional (the
## Medium's evil bit is already in the mask), not the other modifier holders: keying on
## all holders split every class ~12,000 ways in late games.
## walkers: sleepwalk_claimants(). A "known to sleepwalk" claim reads only whether its
## claimants are the Sleepwalker, so only their bits join the mask (0 when nobody claims).
static func coarse_mask(people: int, world: Dictionary, seer: bool, mediums: int, walkers: int = 0) -> int:
	var n: int = people + 1
	var mask: int = (int(world.evil) * n + int(world.get("demon", -1)) + 1) * n + (int(world.marked) + 1 if seer else 0)
	if mediums != 0:
		var deluded: int = int(world.get("holders", 0)) & mediums if world.get("modifier", "marked") == "delusional" else 0
		mask = mask * (1 << people) + deluded
	if walkers != 0:
		var asleep: int = int(world.get("holders", 0)) & walkers if world.get("modifier", "marked") == "sleepwalker" else 0
		mask = mask * (1 << people) + asleep
	if world.has("converted"):
		mask = mask * n + int(world.converted) + 1
	return mask

## Bit per person who said they've been known to sleepwalk (unquoted secret claims).
static func sleepwalk_claimants(ids: Array, cards: Array) -> int:
	var bits: int = 0
	for card in cards:
		if card.get("kind", "") == "secret_claim" and card.get("data", {}).get("secret_kind", "") == "sleepwalker" and not card.get("quoted", false):
			var i: int = ids.find(card.get("subject", ""))
			if i >= 0:
				bits |= 1 << i
	return bits

## Bit per Medium named by an unquoted relay card (the people the relay check reads).
static func relay_mediums(ids: Array, cards: Array) -> int:
	var bits: int = 0
	for card in cards:
		if card.has("relay_source") and not card.get("quoted", false):
			var medium: int = ids.find(card.relay_source)
			if medium >= 0:
				bits |= 1 << medium
	return bits

## Cards the role check reads: testimony-only cards are weighed elsewhere.
static func _live_cards(cards: Array) -> Array:
	return cards.filter(func(card): return not card.get("testimony_only", false))

## The cards routing reads in a world that passes the relay checks: each unquoted relay
## adds its Medium's inferred claim, in card order (as _fits_view does).
static func _routing_cards(cards: Array) -> Array:
	var out: Array = cards.duplicate()
	for card in cards:
		if card.has("relay_source") and not card.get("quoted", false):
			out.append({"kind": "claim", "source": card.relay_source, "role": "medium", "inferred": true})
	return out

## The seer flag fits() keys its memo on.
static func seer_flag(cards: Array) -> bool:
	return cards.any(func(c): return c.get("role") in ["seer", "gravedigger", "martyr"] and not c.get("quoted", false))

## routes: Activity.routes(view, cards, answers, knowledge) if already computed. It
## doesn't depend on the world, so fits_table computes it once for all worlds.
static func _fits_view(view: Dictionary, ids: Array, world: Dictionary, cards: Array, knowledge: Dictionary, answers: Dictionary, routes = null, bluffs: bool = false) -> bool:
	cards = _live_cards(cards)
	# "I've been known to sleepwalk": true of the one Sleepwalker; anyone else saying it
	# is evil and lying. (Two claimants: at least one is evil.)
	var walkers: int = sleepwalk_claimants(ids, cards)
	if walkers != 0:
		var asleep: int = int(world.get("holders", 0)) if world.get("modifier", "marked") == "sleepwalker" else 0
		if walkers & ~int(world.evil) & ~asleep:
			return false
	var visits = {}
	for card in cards:
		if card.has("relay_source") and not card.get("quoted", false):
			var medium: int = ids.find(card.relay_source)
			if medium < 0 or world.evil & (1 << medium) or not preload("res://workshop/world_domain.gd").healthy(view, world, card.relay_source) or not preload("res://workshop/live_history.gd").alive_at_night(view, card.relay_source, card.relay_observed_at.day):
				return false
			var visit: String = card.relay_source + ":" + card.source
			if visits.has(visit) and visits[visit] != card.relay_observed_at.day:
				return false
			visits[visit] = card.relay_observed_at.day
	cards = _routing_cards(cards)
	var sexton_night = -1
	for card in cards:
		if card.kind != "report" or card.get("quoted", false):
			continue
		if card.role == "martyr" and card.data.get("evil", false) and not preload("res://workshop/live_history.gd").executed(view, card.data.target, card.observed_at.day):
			return false
		if card.role == "sexton":
			if sexton_night >= 0 and sexton_night != card.data.night:
				return false
			sexton_night = card.data.night
	if sexton_night >= 0:
		if view.people.filter(func(p): return preload("res://workshop/live_history.gd").alive_at_night(view, p.id, sexton_night)).size() >= 5:
			return false
		if view.get("timeline", {}).get("events", []).any(func(e): return e.kind in ["demon_death", "demon_attack"] and e.at.day == sexton_night):
			return false
		for card in cards:
			if card.get("quoted", false):
				continue
			if card.kind == "report" and card.role in ["nightwatch", "lamplighter", "medium"] and card.get("night", -1) == sexton_night:
				return false
			if card.has("relay_source") and card.relay_observed_at.day == sexton_night:
				return false
	for card in cards:
		if card.get("role") != "veteran" or card.kind != "report" or card.get("quoted", false):
			continue
		var demon: int = world.get("demon", -1)
		if demon < 0:
			return false
		var attack_routes: Array = Activity.demon_routes(view.people[demon].home, view.people[ids.find(card.source)].home)
		var fits_approach: bool = attack_routes.any(func(route): return (route.path[route.path.size() - 2] if route.path.size() >= 2 else view.people[ids.find(card.source)].home) == card.data.block)
		if not fits_approach or view.get("timeline", {}).get("events", []).any(func(e): return e.kind in ["demon_death", "demon_attack"] and e.at.day == card.night):
			return false
	var live: bool = bluffs
	if Activity.available(view):
		if routes == null:
			routes = Activity.routes(view, cards, answers, knowledge)
		# Live testimony can be false (an evil fake sighting, a lying alibi): routes that
		# can't all hold no longer void every world; the movement likelihood weighs them.
		if not routes.okay and not live:
			return false
		if routes.okay:
			cards = cards.duplicate()
			for person in routes.forced_lamp:
				if world.get("demon", -1) == ids.find(person):
					continue # The killer may have taken a detour; no forced good-gift proof.
				cards.append({"kind": "claim", "source": person, "role": "lamplighter", "inferred": true})
	return _fits(ids, world, cards, knowledge, answers, live, view)

## bluffs (live play): in a world where a claim's source is evil, the claim is a bluff:
## it occupies no role and constrains nothing. A good source's claim stays binding, so a
## contradiction leaves only the worlds where some claimant is lying.
static func _fits(ids: Array, world: Dictionary, cards: Array, knowledge: Dictionary, answers: Dictionary, bluffs: bool = false, view: Dictionary = {}) -> bool:
	var all: int = (1 << ids.size()) - 1
	var good: int = all & ~world.evil
	var demon: int = world.get("demon", -1)
	var minions: int = world.evil & ~(1 << demon) if demon >= 0 else world.evil
	var assignments = {}
	var own: int = ids.find(knowledge.get("self", ""))
	if own >= 0 and knowledge.get("own_role", "") in C.ROLES:
		assignments[knowledge.own_role] = 1 << own
	for person in knowledge.get("public_roles", {}):
		var index: int = ids.find(person)
		var role: String = knowledge.public_roles[person]
		if index < 0 or world.evil & (1 << index) or not _assign(assignments, role, 1 << index):
			return false
	var seer = ""
	var veteran_night = -1
	var martyr_target = ""
	var rh: int = good
	for card in cards:
		if card.get("quoted", false):
			continue
		if card.kind == "totem":
			var candidates = 0
			for id in card.data.candidates:
				var index: int = ids.find(id)
				if index >= 0:
					candidates |= 1 << index
			if not _assign(assignments, "revenant", candidates & minions):
				return false
		if card.kind == "event" and card.get("event") == "martyr":
			var actor: int = ids.find(card.data.martyr)
			var saved: int = ids.find(card.data.saved)
			if actor < 0 or saved < 0 or world.evil & ((1 << actor) | (1 << saved)) or not _assign(assignments, "martyr", 1 << actor):
				return false
		if card.kind not in ["report", "claim"]:
			continue
		var source: int = ids.find(card.source)
		if source < 0:
			return false
		var source_evil: bool = (world.evil & (1 << source)) != 0
		if world.has("converted") and not view.is_empty():
			var time = preload("res://workshop/chronology.gd")
			source_evil = time.evil_at(view, world, card.source, time.statement_time(card))
		if source_evil:
			if bluffs and not card.get("inferred", false):
				continue
			return false
		if not _assign(assignments, card.role, 1 << source):
			return false
		if card.kind == "claim":
			continue
		var pair_mask = 0
		for person in card.data.get("pair", []):
			var i: int = ids.find(person)
			if i < 0:
				return false
			pair_mask |= 1 << i
		var answer: Dictionary = answers.get(card.id, {})
		var selected = 0
		for person in answer.get("positive", []):
			var i: int = ids.find(person)
			if i < 0:
				return false
			selected |= 1 << i
		if card.role in ["clerk", "archivist"]:
			var candidates: int = pair_mask & (good if card.role == "clerk" else minions)
			if selected != 0:
				candidates &= selected
			if not _assign(assignments, card.data.role, candidates):
				return false
		elif card.role == "veteran":
			if veteran_night >= 0 and veteran_night != card.night:
				return false
			veteran_night = card.night
		elif card.role == "martyr":
			if martyr_target != "" and martyr_target != card.data.target:
				return false
			martyr_target = card.data.target
			if card.data.get("evil", false):
				var target: int = ids.find(card.data.target)
				if target < 0 or not (world.evil & (1 << target)) and world.marked != target:
					return false
		elif card.role == "gravedigger":
			var target: int = ids.find(card.data.executed)
			if target < 0:
				return false
			var read_role: String = card.data.role
			var evil_role: bool = C.ROLES.get(read_role, {}).get("align") == "evil"
			# Marked can receive any evil-role result without actually holding it.
			if evil_role and world.marked == target:
				continue
			var candidates: int = (1 << demon) if read_role == "demon" and demon >= 0 else (minions if evil_role else good)
			if not _assign(assignments, read_role, (1 << target) & candidates):
				return false
		elif card.role == "seer":
			if seer != "" and seer != card.source:
				return false
			seer = card.source
			rh &= ~(1 << source) # native setup excludes the Seer
			if not card.data.yes:
				if demon >= 0 and pair_mask & (1 << demon):
					return false
				rh &= ~pair_mask
				continue
			var ping: int = selected if selected != 0 else pair_mask
			var reason: String = answer.get("registration", "")
			var demon_ok: bool = demon >= 0 and (ping & (1 << demon)) != 0
			var marked_ok: bool = world.marked >= 0 and (ping & (1 << world.marked)) != 0
			if reason == "demon" and not demon_ok:
				return false
			if reason == "marked" and not marked_ok:
				return false
			if reason == "red_herring" or (reason == "" and not demon_ok and not marked_ok):
				rh &= ping
	if seer != "" and rh == 0:
		return false
	return _matching(assignments.values(), 0)

static func _assign(assignments: Dictionary, role: String, candidates: int) -> bool:
	if assignments.has(role):
		candidates &= assignments[role]
	assignments[role] = candidates
	return candidates != 0

static func _matching(masks: Array, occupied: int) -> bool:
	if masks.is_empty():
		return true
	var remaining: Array = masks.duplicate()
	var available: int = int(remaining.pop_back()) & ~occupied
	while available:
		var bit: int = available & -available
		if _matching(remaining, occupied | bit):
			return true
		available &= ~bit
	return false

## Per world: BLUFF_WEIGHT for each distinct (source, role) declared by a source who is
## evil in that world. Inferred and quoted declarations don't count.
static func bluff_weights(view: Dictionary, universe: Array, cards: Array) -> PackedFloat64Array:
	if view.get("chronology", {}).has("curse"):
		return _temporal_bluff_weights(view, universe, cards)
	if cards.any(func(c): return c.has("hearing_weight") and c.kind in ["claim", "report"]):
		return _temporal_bluff_weights(view, universe, cards)
	var ids: Array = view.people.map(func(p): return p.id)
	var declared: Array = []  # source bit per distinct declaration
	var seen = {}
	for card in cards:
		if card.kind not in ["report", "claim"] or card.get("quoted", false) or card.get("inferred", false) or card.get("testimony_only", false):
			continue
		var i: int = ids.find(card.source)
		var key: String = card.source + "|" + str(card.get("role", ""))
		if i < 0 or seen.has(key):
			continue
		seen[key] = true
		declared.append(1 << i)
	var out = PackedFloat64Array()
	out.resize(universe.size())
	for w in range(universe.size()):
		var evil: int = universe[w].evil
		var weight = 1.0
		for bit in declared:
			if evil & bit:
				weight *= BLUFF_WEIGHT
		out[w] = weight
	return out

static func _temporal_bluff_weights(view: Dictionary, universe: Array, cards: Array) -> PackedFloat64Array:
	var time = preload("res://workshop/chronology.gd")
	var declarations = {}
	for card in cards:
		if card.kind in ["report", "claim"] and not card.get("quoted", false) and not card.get("inferred", false) and not card.get("testimony_only", false):
			var key: String = time.source_key(view, card) + ":" + card.role
			if not declarations.has(key) or float(card.get("hearing_weight", 1.0)) > float(declarations[key].get("hearing_weight", 1.0)):
				declarations[key] = card
	# Per declaration, once: the source's index, statement time and factor. A world with
	# no conversion is evil at every time exactly when its evil bit is set.
	var ids: Array = view.people.map(func(p): return p.id)
	var who = PackedInt32Array()
	var factor = PackedFloat64Array()
	var times: Array = []
	for card in declarations.values():
		who.append(ids.find(card.source))
		factor.append(lerpf(1.0, BLUFF_WEIGHT, float(card.get("hearing_weight", 1.0))))
		times.append(time.statement_time(card))
	var out = PackedFloat64Array()
	out.resize(universe.size())
	for i in range(universe.size()):
		var world: Dictionary = universe[i]
		var converted: bool = world.has("converted")
		var evil: int = int(world.evil)
		var weight = 1.0
		for d in range(who.size()):
			var j: int = who[d]
			if j < 0:
				continue
			if (evil & (1 << j)) != 0 or (converted and time.evil_index(view, world, j, times[d])):
				weight *= factor[d]
		out[i] = weight
	return out

## Case evaluation (docs/CASE_REASONING.md, foundation pass).
##
## One pass over the world universe instead of the old 2^u credibility mixture. Each
## world gets a weight: the listener's prior times, for every report card, its likelihood
## in that world:
##   source can't give a healthy reading in this world (evil, impaired): q
##   otherwise: (1 - err) * [content holds] + err * q
## q is the chance of that particular content arising by chance (a false report), and err
## the chance a healthy reading from a good source is still wrong anyway (unmodelled
## impairment). Evil sources are part of the world, so a liar's several readings are wrong
## together without a separate honesty factor, and with independent errors this sum is
## exactly what enumerating every "which cards are true" combination would give.
##
## Chosen answers are the player's theory: they're scored for fit, never added as facts.
## Hard constraints that aren't reports (movement, alibis, claims, roles) stay with the
## solver's existing checks for now.
##
## Speed: compile() groups the universe once per presentation into classes that every
## listener weighs alike (same evil team, role-check mask, card states and premise), so
## each listener's work is per class (a few hundred), not per world (15,120 live).
## Report tables are per reading and cached, so they're built once, when a reading first
## appears in a case. tools/profile_case_eval_town.gd measures the town.
extends RefCounted

const Solver = preload("res://workshop/solver.gd")
const Domain = preload("res://workshop/world_domain.gd")
const Graph = preload("res://workshop/case_graph.gd")
const TimeModel = preload("res://workshop/chronology.gd")
const Movement = preload("res://workshop/movement_model.gd")

## Chance a healthy reading from a good source is wrong anyway (unmodelled impairment).
## Kept small: the rules make a good, healthy reading accurate, and the failures the
## rules allow (evil, Delusional, Marked registering) are in the worlds themselves. A
## larger value lets "it was just wrong" explain away any surprising reading.
const DEFAULT_ERR = 0.01
## The same for the listener's own reading: they know what they saw.
const OWN_ERR = 0.002
## Likelihood-ratio thresholds for the case-strength categories (neutral listener).
const LR_WEAK = 1.5
const LR_MODERATE = 3.0
const LR_STRONG = 10.0

## Chance of this report's content arising by chance: roughly one over the number of
## different things the gift could have said.
static func chance_content(view: Dictionary, card: Dictionary) -> float:
	var d: Dictionary = card.get("data", {})
	match card.get("role", ""):
		"surveyor":
			var blocks = {}
			for p in view.people:
				blocks[p.home] = true
			return 1.0 / float(blocks.size() + 1)
		"dowser", "hearthwitch":
			return 1.0 / float(d.get("group", []).size() + 1)
		"lamplighter":
			return 1.0 / float(view.people.size())
		"nightwatch":
			return 0.25
	return 0.5

static var _valid_cache: Dictionary = {}

## Per world: can this card's source give a healthy reading? (cached per card and universe)
static func _valid_table(view: Dictionary, card: Dictionary) -> PackedByteArray:
	var key = hash([Solver._universe_key(view), card.kind, card.source, card.observed_at, card.get("role", ""), view.get("timeline", {}), card.get("testimony_only", false), card.get("reported_at")])
	if _valid_cache.has(key):
		return _valid_cache[key]
	if _valid_cache.size() > 512:
		_valid_cache.clear()
	_valid_cache[key] = Solver.valid_table(view, card)
	return _valid_cache[key]

static var _index_cache: Dictionary = {}

## Per universe: the distinct evil teams, and which team each world has (cached).
static func _index(view: Dictionary) -> Dictionary:
	var key = Solver._universe_key(view)
	if _index_cache.has(key):
		return _index_cache[key]
	if _index_cache.size() > 16:
		_index_cache.clear()
	var universe: Array = Solver.universe(view)
	var teams: Array = []
	var team_idx = {}
	var team_of = PackedInt32Array()
	team_of.resize(universe.size())
	for i in range(universe.size()):
		var evil: int = universe[i].evil
		if universe[i].has("converted") and TimeModel.evil_index(view, universe[i], universe[i].converted, "Current alignment"):
			evil |= 1 << int(universe[i].converted)
		if not team_idx.has(evil):
			team_idx[evil] = teams.size()
			teams.append(evil)
		team_of[i] = team_idx[evil]
	var out = {"teams": teams, "team_of": team_of}
	_index_cache[key] = out
	return out

## Uniform prior over legal worlds, with no private knowledge: the reference listener.
static func neutral_listener(view: Dictionary) -> Dictionary:
	return {"id": "", "suspicion": {}, "knowledge": {}, "reading_reliability": {}}

static var _premise_cache: Dictionary = {}

static func _premise_mask(view: Dictionary, conclusion: Dictionary) -> PackedByteArray:
	var key = hash([Solver._universe_key(view), Solver.question_inputs(view, conclusion), view.get("timeline", {})])
	if _premise_cache.has(key):
		return _premise_cache[key]
	if _premise_cache.size() > 256:
		_premise_cache.clear()
	_premise_cache[key] = _build_premise_mask(view, conclusion)
	return _premise_cache[key]

static func _build_premise_mask(view: Dictionary, conclusion: Dictionary) -> PackedByteArray:
	if conclusion.type in ["misled", "unreliable"]:
		return Solver.question_table(view, conclusion)
	var universe: Array = Solver.universe(view)
	var m = PackedByteArray()
	m.resize(universe.size())
	var bits = Solver.world_bits(view)
	var index: int = Solver.person_ids(view).find(conclusion.get("target", ""))
	if bits.plain and index >= 0 and conclusion.get("type", "") in ["evil", "suspicious", "good"]:
		# No conversions: alignment is the world's evil bit.
		var bit: int = 1 << index
		var evil: PackedInt64Array = bits.evil
		var want: int = 0 if conclusion.type == "good" else 1
		for i in range(universe.size()):
			m[i] = want if evil[i] & bit else 1 - want
		return m
	for i in range(universe.size()):
		m[i] = 1 if Solver.holds(view, universe[i], conclusion) else 0
	return m

static func _share(weights: PackedFloat64Array, mask: PackedByteArray) -> Array:
	var yes = 0.0
	var total = 0.0
	for i in range(weights.size()):
		total += weights[i]
		if mask[i] == 1:
			yes += weights[i]
	return [yes, total]

## The chance a reading is wrong despite a good, healthy source, for this listener:
## their reliability for it (impairment, a doubted route, a challenge they accept),
## starting from DEFAULT_ERR, or OWN_ERR for their own reading. A reading their own
## knowledge disowns says nothing.
static func reading_error(view: Dictionary, card: Dictionary, listener: Dictionary, own_ids: Array = []) -> float:
	var rel: Dictionary = listener.get("reading_reliability", {})
	var r = 1.0
	for key in [card.get("accuracy_origin", ""), Solver.semantic_key(card), card.get("fact_id", ""), card.id]:
		if key != "" and rel.has(key):
			r = float(rel[key])
			break
	if (card.has("relay_source") or card.has("route_origin")) and rel.has(card.id) and card.get("accuracy_origin", "") != card.id:
		r *= float(rel[card.id])  # the hearing route itself
	r *= float(listener.get("challenge_reports", {}).get(card.id, 1.0)) * float(listener.get("challenge_sources", {}).get(card.get("source", ""), 1.0))
	if card.kind == "report" and TimeModel.disowned(view, card, listener.get("knowledge", {}).get("known_events", [])):
		r = 0.0
	var base_err: float = OWN_ERR if card.id in own_ids else DEFAULT_ERR
	return clampf(1.0 - clampf(r, 0.0, 1.0) * (1.0 - base_err), OWN_ERR, 1.0)

## How far the evidence moved a probability toward certainty (+1: from p0 all the way
## to yes; -1: all the way to no; 0: no change).
static func shift(p0: float, p1: float) -> float:
	if p1 >= p0:
		return (p1 - p0) / (1.0 - p0) if p0 < 1.0 else 0.0
	return (p1 - p0) / p0 if p0 > 0.0 else 0.0

static func _odds(p: float) -> float:
	p = clampf(p, 1e-9, 1.0 - 1e-9)
	return p / (1.0 - p)

## Strength category from a likelihood ratio on the premise.
static func category(lr: float) -> String:
	if lr < 1.0 / LR_WEAK:
		return "contradicts"
	if lr < LR_WEAK:
		return "none"
	if lr < LR_MODERATE:
		return "weak"
	if lr < LR_STRONG:
		return "moderate"
	return "strong"

static func category_text(cat: String) -> String:
	match cat:
		"contradicts": return "The linked evidence argues against this premise."
		"none": return "The linked evidence doesn't support this premise."
		"weak": return "This case weakly supports the premise."
		"moderate": return "This case moderately supports the premise."
		"strong": return "This case strongly supports the premise."
	return ""

## Evaluate a case for one listener. Returns:
##   prior, posterior    P(premise) before and after the case's evidence
##   lr, category        likelihood ratio on the premise and its category
##   people              per person {prior, posterior} probability of being evil
##   rivals              the people (other than the target) most likely evil after the case
##   theory_fit          how much of the posterior agrees with the player's answers/positions
##   pivotal             card ids whose removal would drop the category (reference listener)
##   impossible          no world survives the case's hard facts
static func evaluate(view: Dictionary, case_data: Dictionary, listener: Dictionary = {}) -> Dictionary:
	if listener.is_empty():
		listener = neutral_listener(view)
	# An already-compiled (attended) case has its lines folded away: keep its factors.
	var supplied: Array = case_data.get("test_factors", [])
	case_data = Graph.compile_case(view, case_data)
	if case_data.test_factors.is_empty() and not supplied.is_empty():
		case_data.test_factors = supplied.duplicate(true)
	var cards: Array = Solver.selected(view, case_data)
	var conclusion: Dictionary = case_data.conclusion
	var knowledge: Dictionary = listener.get("knowledge", {})
	var own_ids: Array = knowledge.get("own_reports", [])
	var own: Array = own_ids.map(func(id): return Solver.card_by_id(view, id)).filter(func(c): return not c.is_empty() and not (c in cards) and c.kind == "report" and Solver.tabled(c))
	var err_for = func(card: Dictionary) -> float:
		return reading_error(view, card, listener, own_ids)
	var cc = compile(view, cards, case_data, own)
	var oc = _own_counts(view, cc, own)
	var nc: int = cc.team.size()
	var ns: int = oc.states
	# Per class: prior (team odds, role fits) and the case's likelihood.
	var team_prior = _team_prior(view, listener)
	var fits = _class_fits(view, cc, cards, knowledge)
	# The "before" side of the case: role knowledge without the case's own cards. Role-
	# based readings (Seer, Archivist, Clerk...) act through the role check; applied to
	# both sides they cancel, and the reading never moved anyone (playtest 37291).
	var fits_before = _class_fits(view, cc, [], knowledge)
	if view.get("chronology", {}).has("curse"):
		fits = fits.duplicate() # Never modify a cached role table.
		var worlds: Array = Solver.universe(view)
		for c in range(fits.size()):
			var world: Dictionary = worlds[cc.rep[c]]
			for person in knowledge.get("good", []):
				if TimeModel.evil_at(view, world, person, knowledge.get("good_epoch", view.alignment_epoch)):
					fits[c] = 0.0
			for fact in knowledge.get("timed_good", []):
				if TimeModel.evil_at(view, world, fact.person, fact.at):
					fits[c] = 0.0
	var case_like = _class_like(cc, cc.reports.map(func(c): return err_for.call(c)), -1)
	var own_like = _state_like(view, own, own.map(func(c): return err_for.call(c)), oc.codes)
	# Live role uniqueness is checked per world in Roles.fits, where an evil claimant may
	# bluff; a global "two people claimed one gift" gate would void every world.
	var compatible: bool = true
	var base = PackedFloat64Array()
	var post = PackedFloat64Array()
	base.resize(nc * ns)
	post.resize(nc * ns)
	var counts: PackedFloat64Array = oc.counts
	var team: PackedInt32Array = cc.team
	var move: PackedFloat64Array = cc.move if cc.movement_ok else PackedFloat64Array()
	if not cc.movement_ok:
		move.resize(nc)
		move.fill(0.0)
	var test_like = _test_like(view, cc, listener, own_ids)
	for c in range(nc):
		var w0: float = team_prior[team[c]] * fits[c] if compatible else 0.0
		for o in range(ns):
			var k = c * ns + o
			var b: float = counts[k] * w0 * own_like[o]
			base[k] = counts[k] * team_prior[team[c]] * fits_before[c] * own_like[o] if compatible else 0.0
			post[k] = b * case_like[c] * move[c] * test_like[c]
	# A listener holding a theory reads the evidence through it (Theory bias): worlds
	# that fit their theory count for more after the case.
	var bias: Dictionary = listener.get("theory_bias", {})
	if not bias.is_empty():
		var universe: Array = Solver.universe(view)
		var rep_of: PackedInt32Array = cc.rep
		for c in range(nc):
			if Solver.position_holds(view, universe[rep_of[c]], bias.target, bias.position, conclusion.get("at", "")):
				for o in range(ns):
					post[c * ns + o] *= float(bias.factor)
	var b2 = _class_share(cc, base, ns)
	var a2 = _class_share(cc, post, ns)
	var p0 = b2[0] / b2[1] if b2[1] > 0.0 else 0.0
	var p1 = a2[0] / a2[1] if a2[1] > 0.0 else 0.0
	var lr = _odds(p1) / _odds(p0) if p0 > 0.0 and p0 < 1.0 else 1.0
	var result = {"prior": p0, "posterior": p1, "lr": lr, "category": category(lr),
		"impossible": a2[1] <= 0.0}
	result["people"] = _people(view, cc, base, post, ns)
	var test_used: Array = cc.tests.filter(func(f): return _route_credit(view, f, listener, own_ids) > 0.0)
	var joint = _joint(view, cc, oc, own, base, post, ns, err_for)
	result["consistent"] = joint.consistent
	result["acceptance"] = joint.acceptance
	result["credence"] = joint.credence
	result["movement"] = cc.movement + test_used.map(func(f): return f.label)
	result["factors"] = cc.factors + test_used
	result["rivals"] = _rivals(view, result.people, conclusion)
	result["theory_fit"] = 1.0
	if not case_data.get("positions", {}).is_empty() or case_data.get("answers", {}).values().any(func(a): return not a.get("positive", []).is_empty() or not a.get("blocks", []).is_empty()):
		result["theory_fit"] = _theory_fit(view, case_data, cards, _expand(cc, oc, post))
	if listener.get("id", "") == "":
		# Only the board's strength label needs it: the reference listener.
		result["pivotal"] = _pivotal(cc, base, ns, err_for, p0, result.category)
	return result

static var _compiled: Dictionary = {}

## Per presentation: the universe grouped into classes that every listener weighs alike.
## A class shares the evil team, what the role check reads, each report card's state
## (holds / healthy but wrong / source can't read healthily), the hard event facts and
## the premise. Listener-specific work is then per class, not per world.
## known: the listener's own readings, which can complete a test factor's route.
static func compile(view: Dictionary, cards: Array, case_data: Dictionary, known: Array = []) -> Dictionary:
	var conclusion: Dictionary = case_data.conclusion
	var key_input = [Solver._universe_key(view), view.get("timeline", {}), view.people, conclusion, case_data.get("citations", {}), case_data.get("test_factors", []),
		known.map(func(c): return c.get("fact_id", c.id)) if not case_data.get("test_factors", []).is_empty() else [],
		cards.map(func(c): return [c.get("fact_id", c.id), c.kind, c.source, c.get("role", ""), c.get("observed_at", {}), c.get("data", {}), c.get("event", ""), c.has("relay_source"), c.get("quoted", false), c.get("block", ""), c.get("night", -1), c.get("subject", ""), c.get("subjects", [])])]
	if cards.any(func(c): return c.has("hearing_weight")):
		key_input.append(cards.map(func(c): return c.get("hearing_weight", 1.0)))
	if cards.any(func(c): return c.get("testimony_only", false)):
		key_input.append(cards.map(func(c): return [c.get("testimony_only", false), c.get("reported_at")]))
	var key = hash(key_input)
	if _compiled.has(key) and _compiled[key].input == key_input:
		return _compiled[key]
	if _compiled.size() > 32:
		_compiled.clear()
	var universe: Array = Solver.universe(view)
	var n: int = universe.size()
	var ix = _index(view)
	var reports: Array = []
	var tables: Array = []
	var valids: Array = []
	var dead = PackedByteArray()
	dead.resize(n)
	dead.fill(0)
	for card in cards:
		if not Solver.tabled(card):
			continue
		var table: PackedByteArray = Solver.report_table(view, card)
		if card.kind != "report":
			# A public event (the curse) is a fact, not a fallible reading.
			for i in range(n):
				if table[i] == 0:
					dead[i] = 1
			continue
		reports.append(card)
		tables.append(table)
		valids.append(_valid_table(view, card))
	var premise = _premise_mask(view, conclusion)
	var people: int = view.people.size()
	var roles_on: bool = view.get("world_options", {}).get("roles", false)
	var seer: bool = Solver.Roles.seer_flag(cards)
	var mediums: int = Solver.Roles.relay_mediums(view.people.map(func(p): return p.id), cards)
	var walkers: int = Solver.Roles.sleepwalk_claimants(view.people.map(func(p): return p.id), cards)
	var k: int = reports.size()
	var index = {}
	var team = PackedInt32Array()
	var rep = PackedInt32Array()
	var count = PackedFloat64Array()
	var prem = PackedByteArray()
	var states = PackedByteArray()  # class-major, k per class
	var is_dead = PackedByteArray()
	var class_of = PackedInt32Array()
	class_of.resize(n)
	var digits = PackedByteArray()
	digits.resize(k)
	var radix: int = int(pow(3, k))
	var rm_bound: int = (1 << people) * (people + 2) * (people + 2) * ((1 << people) if mediums != 0 else 1) * ((1 << people) if walkers != 0 else 1)
	if view.get("chronology", {}).has("curse"):
		rm_bound *= people + 1
	var int_keys: bool = k <= 16 and float(rm_bound) * float(radix) * 4.0 < 9.0e18
	# Each world's card states as one base-3 code, a card at a time.
	var codes = PackedInt64Array()
	codes.resize(n)
	codes.fill(0)
	for c in range(k):
		var t: PackedByteArray = tables[c]
		var v: PackedByteArray = valids[c]
		for i in range(n):
			codes[i] = codes[i] * 3 + (0 if t[i] == 1 else (1 if v[i] == 1 else 2))
	var live: bool = Domain.live(view)
	var rms: PackedInt64Array = _coarse(view, seer, mediums, walkers) if roles_on else Solver.world_bits(view).evil
	if not roles_on and view.get("chronology", {}).has("curse"):
		rms = rms.duplicate()
		for i in range(n):
			rms[i] = rms[i] * (people + 1) + int(universe[i].converted) + 1
	if not live:
		# Fixture universes are small and carry conversion histories: one world per class.
		rms = PackedInt64Array()
		rms.resize(n)
		for i in range(n):
			rms[i] = i
	# Movement and trace observations: a likelihood per world from the evil team and
	# the Demon (live) or each subject's alignment and Mark (fixtures). Both are fixed
	# within a class.
	# Live factors are cached arrays. Extending one in place would contaminate
	# later people/cases with repeated public-event likelihoods.
	var movement: Array = Movement.factors(view, cards, case_data).duplicate()
	movement.append_array(preload("res://workshop/public_evidence.gd").factors(view))
	movement.append_array(preload("res://workshop/speech_evidence.gd").factors(cards))
	# Supplied test factors (and a challenged home alibi in fixtures): a likelihood ratio
	# on their target being evil, once any of their routes is among the cards.
	# Each listener weighs them by how far they credit the route (see _factor_like).
	var semantic = {}
	for card in cards + known:
		semantic[Solver.semantic_key(card)] = true
	var tests: Array = []
	for factor in case_data.get("test_factors", []):
		if is_zero_approx(float(factor.log2_ratio)):
			continue
		var routes: Array = factor.routes.filter(func(route): return route.all(func(id): return semantic.has(Solver.semantic_key(Solver.card_by_id(view, id)))))
		if not routes.is_empty():
			var f: Dictionary = factor.duplicate(true)
			f["epoch"] = conclusion.get("at", "")
			f.routes = routes
			tests.append(f)
	var test_hold = PackedByteArray()  # class-major: does each test factor's target fit?
	var move = PackedFloat64Array()
	var team_of: PackedInt32Array = ix.team_of
	var plain_bits: bool = Solver.world_bits(view).plain
	var group = PackedInt32Array()  # per class: index into group_bits (who is evil then)
	var group_bits: Array = ix.teams if plain_bits else []
	var group_index = {}
	for i in range(n):
		var code: int = codes[i]
		var rm: int = rms[i]
		# One int key when it fits in 63 bits, else an array.
		var ck = ((rm * radix + code) * 2 + premise[i]) * 2 + dead[i] if int_keys else [rm, code, premise[i], dead[i]]
		var c_id = index.get(ck, -1)
		if c_id < 0:
			c_id = team.size()
			index[ck] = c_id
			team.append(team_of[i])
			if plain_bits:
				group.append(team_of[i])
			else:
				# Conversion histories: who is evil at the case's epoch, not originally.
				var bits = 0
				for j in range(people):
					if TimeModel.evil_index(view, universe[i], j, conclusion.get("at", view.get("alignment_epoch", ""))):
						bits |= 1 << j
				var g_id = group_index.get(bits, -1)
				if g_id < 0:
					g_id = group_bits.size()
					group_index[bits] = g_id
					group_bits.append(bits)
				group.append(g_id)
			rep.append(i)
			count.append(0.0)
			prem.append(premise[i])
			is_dead.append(dead[i])
			move.append(_movement_score(view, universe[i], movement))
			for f in tests:
				var target: int = Solver.person_index(view, f.target)
				test_hold.append(1 if target >= 0 and TimeModel.evil_at(view, universe[i], f.target, f.epoch) else 0)
			for c in range(k - 1, -1, -1):
				digits[c] = code % 3
				code /= 3
			states.append_array(digits)
		count[c_id] += 1.0
		class_of[i] = c_id
	var out = {"input": key_input, "key": key, "reports": reports, "tables": tables, "valids": valids, "move": move,
		"movement": movement.map(func(f): return f.label), "factors": movement, "tests": tests, "test_hold": test_hold, "group": group, "group_bits": group_bits, "movement_ok": _movement_ok(view, cards, live),
		"q": reports.map(func(c): return chance_content(view, c)), "team": team, "rep": rep, "count": count,
		"premise": prem, "dead": is_dead, "states": states, "class_of": class_of, "k": k}
	for factor in movement:
		out.movement.append_array(factor.get("relationships", []))
	_compiled[key] = out
	return out

static var _coarse_cache: Dictionary = {}

## Per universe: each world's role-check mask (Roles.coarse_mask), for these flags.
static func _coarse(view: Dictionary, seer: bool, mediums: int, walkers: int = 0) -> PackedInt64Array:
	var key = hash([Solver._universe_key(view), seer, mediums, walkers])
	if _coarse_cache.has(key):
		return _coarse_cache[key]
	if _coarse_cache.size() > 16:
		_coarse_cache.clear()
	var universe: Array = Solver.universe(view)
	var people: int = view.people.size()
	var out = PackedInt64Array()
	out.resize(universe.size())
	for i in range(universe.size()):
		out[i] = Solver.Roles.coarse_mask(people, universe[i], seer, mediums, walkers)
	_coarse_cache[key] = out
	return out

## A world's movement/trace likelihood ratio (good everywhere = 1), as the support
## model's world_score without its prior ranks.
static func _movement_score(view: Dictionary, world: Dictionary, factors: Array) -> float:
	var score = 1.0
	for factor in factors:
		if factor.get("joint_live", false):
			score *= Movement.Live.score(world, factor)
			continue
		var target: int = Solver.person_index(view, factor.target)
		if target >= 0 and TimeModel.evil_at(view, world, factor.target, factor.get("epoch", view.get("alignment_epoch", ""))):
			score *= pow(2.0, float(factor.log2_ratio))
		elif target >= 0 and factor.get("origin", "") == "movement_model" and world.marked == target:
			score *= maxf(factor.good_marked_likelihood, 1e-12) / maxf(factor.good_likelihood, 1e-12)
	return score

## Fixture views: sightings and alibis that can't all be true leave no consistent world.
## (Live views check routes inside the role check.)
static func _movement_ok(view: Dictionary, cards: Array, live: bool) -> bool:
	if live:
		return true
	return Solver.solve(view, cards).get("movement_ok", true)

## How far the listener credits a test factor's best route: the product over its cards
## of each reading's reliability (0 when they reject one outright).
static func _route_credit(view: Dictionary, factor: Dictionary, listener: Dictionary, own_ids: Array) -> float:
	var best = 0.0
	for route in factor.routes:
		var credit = 1.0
		for id in route:
			var card: Dictionary = Solver.card_by_id(view, id)
			if card.kind == "report":
				credit *= (1.0 - reading_error(view, card, listener, own_ids)) / (1.0 - (OWN_ERR if id in own_ids else DEFAULT_ERR))
			elif listener.get("reading_reliability", {}).has(Solver.semantic_key(card)):
				credit *= float(listener.reading_reliability[Solver.semantic_key(card)])
		best = maxf(best, clampf(credit, 0.0, 1.0))
	return best

## Test factors per class: 2^log2_ratio where the target is evil, mixed with no
## effect by how far the listener credits the route.
static func _test_like(view: Dictionary, cc: Dictionary, listener: Dictionary, own_ids: Array) -> PackedFloat64Array:
	var nc: int = cc.team.size()
	var out = PackedFloat64Array()
	out.resize(nc)
	out.fill(1.0)
	var tests: Array = cc.tests
	var m: int = tests.size()
	if m == 0:
		return out
	var hold: PackedByteArray = cc.test_hold
	for j in range(m):
		var w: float = _route_credit(view, tests[j], listener, own_ids)
		var ratio: float = pow(2.0, float(tests[j].log2_ratio))
		for c in range(nc):
			if hold[c * m + j] == 1:
				out[c] *= (1.0 - w) + w * ratio
	return out

static var _own_cache: Dictionary = {}

## Class counts refined by the listener's own private readings' states (3^own states).
## Depends only on which own cards there are, so listeners holding the same share it.
static func _own_counts(view: Dictionary, cc: Dictionary, own: Array) -> Dictionary:
	if own.is_empty():
		return {"states": 1, "counts": cc.count, "of": PackedInt32Array(), "codes": [PackedByteArray()]}
	var key = hash([cc.key, own.map(func(c): return c.get("fact_id", c.id))])
	if _own_cache.has(key):
		return _own_cache[key]
	if _own_cache.size() > 64:
		_own_cache.clear()
	var tables: Array = own.map(func(c): return Solver.report_table(view, c))
	var valids: Array = own.map(func(c): return _valid_table(view, c))
	var class_of: PackedInt32Array = cc.class_of
	var n: int = class_of.size()
	var of = PackedInt32Array()
	of.resize(n)
	# Only states actually present in legal worlds can carry probability. Enumerating
	# all 3^private-readings combinations allocated mostly zero cells and made later
	# presentations exponentially slower. Byte-vector keys also avoid integer overflow.
	var codes: Array = []
	var indices = {}
	var k: int = own.size()
	if k <= 39:
		# Each world's state as one base-3 integer; a byte code is built only the first
		# time a state appears, so indices and codes keep first-appearance order.
		for i in range(n):
			var packed: int = 0
			for c in range(k):
				packed = packed * 3 + (0 if tables[c][i] == 1 else (1 if valids[c][i] == 1 else 2))
			if not indices.has(packed):
				var code = PackedByteArray()
				code.resize(k)
				for c in range(k):
					code[c] = 0 if tables[c][i] == 1 else (1 if valids[c][i] == 1 else 2)
				indices[packed] = codes.size()
				codes.append(code)
			of[i] = indices[packed]
	else:
		for i in range(n):
			var code = PackedByteArray()
			code.resize(k)
			for c in range(k):
				code[c] = 0 if tables[c][i] == 1 else (1 if valids[c][i] == 1 else 2)
			if not indices.has(code):
				indices[code] = codes.size()
				codes.append(code)
			of[i] = indices[code]
	var ns = codes.size()
	var counts = PackedFloat64Array()
	counts.resize(cc.team.size() * ns)
	counts.fill(0.0)
	for i in range(n):
		counts[class_of[i] * ns + of[i]] += 1.0
	var out = {"states": ns, "counts": counts, "of": of, "codes": codes}
	_own_cache[key] = out
	return out

## Likelihood factor of one state for a card: holds, healthy but wrong, or no healthy reading.
static func _factor(st: int, err: float, q: float) -> float:
	match st:
		0: return (1.0 - err) + err * q
		1: return err * q
	return q

## The case's likelihood per class. skip >= 0 leaves that report card out (pivotal check).
static func _class_like(cc: Dictionary, errs: Array, skip: int) -> PackedFloat64Array:
	var nc: int = cc.team.size()
	var k: int = cc.k
	var like = PackedFloat64Array()
	like.resize(nc)
	var states: PackedByteArray = cc.states
	var dead: PackedByteArray = cc.dead
	# Three possible factors per card: price them once (a skipped card counts 1).
	var q: Array = cc.q
	var f = PackedFloat64Array()
	f.resize(3 * k)
	for c in range(k):
		for st in range(3):
			f[3 * c + st] = 1.0 if c == skip else _factor(st, errs[c], q[c])
	for x in range(nc):
		if dead[x] == 1:
			like[x] = 0.0
			continue
		var v = 1.0
		var at: int = x * k
		for c in range(k):
			v *= f[3 * c + states[at + c]]
		like[x] = v
	return like

## The likelihood of each own-reading state (decoded from base 3).
static func _state_like(view: Dictionary, own: Array, errs: Array, codes: Array) -> PackedFloat64Array:
	var out = PackedFloat64Array()
	out.resize(codes.size())
	var qs: Array = own.map(func(c): return chance_content(view, c))
	for o in range(codes.size()):
		var v = 1.0
		for c in range(own.size() - 1, -1, -1):
			v *= _factor(codes[o][c], errs[c], qs[c])
		out[o] = v
	return out

## Per team: the listener's odds that exactly this team is evil (known-good people zero).
static func _team_prior(view: Dictionary, listener: Dictionary) -> PackedFloat64Array:
	var ix = _index(view)
	var ids: Array = Solver.person_ids(view)
	var odds: Array = []
	for id in ids:
		var sv: float = clampf(float(listener.get("suspicion", {}).get(id, 0.3)), 0.02, 0.98)
		# Trust in a person halves (full trust) or doubles (none) their odds of evil.
		var trust: float = float(listener.get("npc_trust", {}).get(id, 0.5))
		odds.append(sv / (1.0 - sv) * pow(2.0, -2.0 * (clampf(trust, 0.0, 1.0) - 0.5)))
	var good_mask = 0
	for id in listener.get("knowledge", {}).get("good", []):
		var idx = ids.find(id)
		if idx >= 0:
			good_mask |= 1 << idx
	var team_prior = PackedFloat64Array()
	team_prior.resize(ix.teams.size())
	for t in range(ix.teams.size()):
		var evil: int = ix.teams[t]
		if evil & good_mask:
			team_prior[t] = 0.0
			continue
		var v = 1.0
		for j in range(ids.size()):
			if evil & (1 << j):
				v *= odds[j]
		team_prior[t] = v
	return team_prior

static var _fits_cache: Dictionary = {}

## Role assignment per class (live views): one check per class representative, since
## a class shares everything the role check reads. A listener's own role and outings only
## add constraints, so only the classes the knowledge-free check allows are re-checked.
static func _class_fits(view: Dictionary, cc: Dictionary, cards: Array, knowledge: Dictionary) -> PackedFloat64Array:
	var nc: int = cc.team.size()
	if not view.get("world_options", {}).get("roles", false):
		var ones = PackedFloat64Array()
		ones.resize(nc)
		ones.fill(1.0)
		return ones
	var personal = {"self": knowledge.get("self", ""), "own_role": knowledge.get("own_role", ""), "own_outings": knowledge.get("own_outings", {})}
	if not knowledge.get("public_roles", {}).is_empty():
		personal["public_roles"] = knowledge.public_roles
	var bare: bool = cards.is_empty() # the "before" side: no case cards (see evaluate)
	var neutral_key = hash([cc.key, "neutral", Domain.live(view), bare])
	if not _fits_cache.has(neutral_key):
		_fits_cache[neutral_key] = _fits_over(view, cc, cards, {}, range(nc))
	var neutral: PackedFloat64Array = _fits_cache[neutral_key]
	if personal.own_role == "" and personal.own_outings.is_empty() and personal.get("public_roles", {}).is_empty():
		return neutral
	var key = hash([cc.key, personal, Domain.live(view), bare])
	if not _fits_cache.has(key):
		if _fits_cache.size() > 256:
			_fits_cache.clear()
			_fits_cache[neutral_key] = neutral
		var open: Array = []
		for c in range(nc):
			if neutral[c] > 0.0:
				open.append(c)
		_fits_cache[key] = _fits_over(view, cc, cards, personal, open)
	return _fits_cache[key]

static func _fits_over(view: Dictionary, cc: Dictionary, cards: Array, knowledge: Dictionary, classes) -> PackedFloat64Array:
	var out = PackedFloat64Array()
	out.resize(cc.team.size())
	out.fill(0.0)
	var universe: Array = Solver.universe(view)
	var reps: Array = []
	var which: Array = []
	for c in classes:
		reps.append(universe[cc.rep[c]])
		which.append(c)
	var live: bool = Domain.live(view)
	var t: PackedByteArray = Solver.Roles.fits_table(view, reps, cards, knowledge, {}, live)
	# Only classes that fail the content check need the declaration-only check.
	var declarations = PackedByteArray()
	if live and t.has(0) and cards.any(func(card): return card.kind == "report"):
		var claims: Array = []
		for card in cards:
			var claim: Dictionary = card
			if card.kind == "report":
				claim = card.duplicate()
				claim.kind = "claim"
				claim.data = {}
			claims.append(claim)
		var failed: Array = []
		var at: PackedInt32Array = PackedInt32Array()
		for x in range(reps.size()):
			if t[x] == 0:
				failed.append(reps[x])
				at.append(x)
		var sub: PackedByteArray = Solver.Roles.fits_table(view, failed, claims, knowledge, {}, true)
		declarations.resize(reps.size())
		declarations.fill(0)
		for y in range(at.size()):
			declarations[at[y]] = sub[y]
	var weights: PackedFloat64Array = Solver.Roles.bluff_weights(view, reps, cards) if live else PackedFloat64Array()
	for x in range(which.size()):
		var content: float = float(t[x])
		if content == 0.0 and not declarations.is_empty() and declarations[x] != 0:
			# A joint report-content contradiction can be an erroneous reading too.
			# Identity/public constraints remain hard; never erase the whole view
			# merely because several claimed Seer results share no exact Red Herring.
			content = DEFAULT_ERR
		out[which[x]] = content * (weights[x] if live else 1.0)
	return out

## Joint facts about the reports, for one listener:
##   consistent   every report could hold together in a world the listener allows
##   acceptance   per report (case and own), the posterior chance the reading is accurate
##   credence     the prior chance every case report's source could read healthily, times
##                each reading's own accuracy: how far a listener who looks for no rival
##                explanation takes the reports on trust
static func _joint(view: Dictionary, cc: Dictionary, oc: Dictionary, own: Array, base: PackedFloat64Array, post: PackedFloat64Array, ns: int, err_for: Callable) -> Dictionary:
	var k: int = cc.k
	var ko: int = own.size()
	var states: PackedByteArray = cc.states
	var dead: PackedByteArray = cc.dead
	var move: PackedFloat64Array = cc.move
	# A reading that holds was accurate with chance (1 - err) / ((1 - err) + err q).
	var accurate = PackedFloat64Array()
	var cards: Array = cc.reports + own
	for card in cards:
		var err: float = err_for.call(card)
		var q: float = chance_content(view, card)
		accurate.append((1.0 - err) / maxf((1.0 - err) + err * q, 1e-12))
	var held = PackedFloat64Array()
	held.resize(k + ko)
	held.fill(0.0)
	var total = 0.0
	var base_total = 0.0
	var readable = 0.0
	var consistent = false
	for x in range(base.size()):
		var c: int = x / ns
		var all_hold = dead[c] == 0 and move[c] > 0.0 and cc.movement_ok
		var all_valid = true
		for j in range(k):
			var st: int = states[c * k + j]
			if st == 0:
				held[j] += post[x]
			else:
				all_hold = false
			if st == 2:
				all_valid = false
		if ko > 0:
			var digits: PackedByteArray = oc.codes[x % ns]
			for j in range(ko):
				if digits[j] == 0:
					held[k + j] += post[x]
		total += post[x]
		base_total += base[x]
		if all_valid:
			readable += base[x]
		if all_hold and base[x] > 0.0:
			consistent = true
	var acceptance = {}
	for j in range(k + ko):
		acceptance[cards[j].id] = held[j] / total * accurate[j] if total > 0.0 else 0.0
	var credence: float = readable / base_total if base_total > 0.0 else 0.0
	for card in cc.reports:
		credence *= minf(1.0, (1.0 - err_for.call(card)) / (1.0 - DEFAULT_ERR))
	return {"consistent": consistent, "acceptance": acceptance, "credence": credence}

static func _class_share(cc: Dictionary, w: PackedFloat64Array, ns: int) -> Array:
	var yes = 0.0
	var total = 0.0
	var prem: PackedByteArray = cc.premise
	for k in range(w.size()):
		total += w[k]
		if prem[k / ns] == 1:
			yes += w[k]
	return [yes, total]

## Per-world weights from class weights (only for checks that read individual worlds).
static func _expand(cc: Dictionary, oc: Dictionary, w: PackedFloat64Array) -> PackedFloat64Array:
	var class_of: PackedInt32Array = cc.class_of
	var ns: int = oc.states
	var counts: PackedFloat64Array = oc.counts
	var of: PackedInt32Array = oc.of
	var out = PackedFloat64Array()
	out.resize(class_of.size())
	for i in range(class_of.size()):
		var k: int = class_of[i] * ns + (of[i] if ns > 1 else 0)
		out[i] = w[k] / counts[k] if counts[k] > 0.0 else 0.0
	return out

static func _people(view: Dictionary, cc: Dictionary, base: PackedFloat64Array, post: PackedFloat64Array, ns: int) -> Dictionary:
	var ids: Array = Solver.person_ids(view)
	var nt: int = cc.group_bits.size()
	var tb = PackedFloat64Array()
	var ta = PackedFloat64Array()
	tb.resize(nt)
	ta.resize(nt)
	tb.fill(0.0)
	ta.fill(0.0)
	var team: PackedInt32Array = cc.group
	for k in range(base.size()):
		var t: int = team[k / ns]
		tb[t] += base[k]
		ta[t] += post[k]
	var sb = 0.0
	var sa = 0.0
	for t in range(nt):
		sb += tb[t]
		sa += ta[t]
	var teams: Array = cc.group_bits
	var pbs = PackedFloat64Array()
	var pas = PackedFloat64Array()
	pbs.resize(ids.size())
	pas.resize(ids.size())
	pbs.fill(0.0)
	pas.fill(0.0)
	for t in range(nt):
		var evil: int = teams[t]
		var j = 0
		while evil:
			if evil & 1:
				pbs[j] += tb[t]
				pas[j] += ta[t]
			evil >>= 1
			j += 1
	var out = {}
	for j in range(ids.size()):
		var pb: float = pbs[j]
		var pa: float = pas[j]
		out[ids[j]] = {"prior": pb / sb if sb > 0.0 else 0.0, "posterior": pa / sa if sa > 0.0 else 0.0}
	return out

static func _rivals(view: Dictionary, people: Dictionary, conclusion: Dictionary, limit: int = 3) -> Array:
	var ids: Array = people.keys().filter(func(id): return id != conclusion.get("target", ""))
	ids.sort_custom(func(x, y): return people[x].posterior > people[y].posterior)
	return ids.slice(0, limit).map(func(id): return {"id": id, "posterior": people[id].posterior, "shift": people[id].posterior - people[id].prior})

## The player's answers and positions as a theory: the share of the posterior consistent
## with them. 1.0 when the player left everything open.
static func _theory_fit(view: Dictionary, case_data: Dictionary, cards: Array, post: PackedFloat64Array) -> float:
	var universe: Array = Solver.universe(view)
	var masks: Array = []
	for card in cards:
		var positive: Array = case_data.get("answers", {}).get(card.id, {}).get("positive", [])
		var blocks: Array = case_data.get("answers", {}).get(card.id, {}).get("blocks", [])
		var marked: Array = case_data.get("answers", {}).get(card.id, {}).get("marked", [])
		if Solver.tabled(card) and (not positive.is_empty() or not blocks.is_empty()):
			masks.append(Solver.report_table(view, card, positive, blocks, marked))
	var positions: Dictionary = case_data.get("positions", {})
	if masks.is_empty() and positions.is_empty():
		return 1.0
	var fit = 0.0
	var total = 0.0
	for i in range(universe.size()):
		total += post[i]
		if post[i] == 0.0:
			continue
		var ok = true
		for m in masks:
			if m[i] == 0:
				ok = false
				break
		if ok:
			for id in positions:
				if not Solver.position_holds(view, universe[i], id, positions[id], case_data.conclusion.get("at", "")):
					ok = false
					break
		if ok:
			fit += post[i]
	return fit / total if total > 0.0 else 0.0

## Cards the case rests on: removing any one of them would drop the strength category.
static func _pivotal(cc: Dictionary, base: PackedFloat64Array, ns: int, err_for: Callable, p0: float, cat: String) -> Array:
	if cat in ["none", "contradicts"]:
		return []
	var reports: Array = cc.reports
	if reports.size() < 2:
		return reports.map(func(c): return c.id)
	var errs: Array = reports.map(func(c): return err_for.call(c))
	var out: Array = []
	for skip in range(reports.size()):
		var like = _class_like(cc, errs, skip)
		var w = base.duplicate()
		for k in range(w.size()):
			w[k] *= like[k / ns]
		var a = _class_share(cc, w, ns)
		var p1 = a[0] / a[1] if a[1] > 0.0 else 0.0
		var lr = _odds(p1) / _odds(p0) if p0 > 0.0 and p0 < 1.0 else 1.0
		if _rank(category(lr)) < _rank(cat):
			out.append(reports[skip].id)
	return out

static func _rank(cat: String) -> int:
	return ["contradicts", "none", "weak", "moderate", "strong"].find(cat)

## The case-strength label for the board: the neutral reference listener's category plus a
## one-line reason. Describes the case, not any particular NPC.
static func strength(view: Dictionary, case_data: Dictionary) -> Dictionary:
	return strength_from(evaluate(view, case_data))

## strength() from an evaluation already in hand.
static func strength_from(r: Dictionary) -> Dictionary:
	var reason = ""
	if r.impossible:
		reason = "These cards can't all be true together; something here is false."
	elif r.category in ["weak", "moderate", "strong"]:
		if r.pivotal.size() == 1:
			reason = "It rests on one card."
		elif r.pivotal.is_empty():
			reason = "Several lines of evidence agree."
		else:
			reason = "It rests on %d cards together." % r.pivotal.size()
	return {"category": r.category, "text": category_text(r.category), "reason": reason, "lr": r.lr}

static func analyze(view: Dictionary, case_data: Dictionary, knowledge: Dictionary = {}, include_conflicts: bool = true) -> Dictionary:
	case_data = Graph.compile_case(view, case_data)
	var cards = selected(view, case_data)
	if case_data.conclusion.type == "conflict":
		cards = cards.filter(func(c): return c.id in Graph.roots(case_data, case_data.conflict_scope) or c.id in knowledge.get("own_reports", []))
	var base = solve(view, cards, {}, {}, knowledge)
	var proposed_cards = TimeModel.hypothetical_cards(view, case_data, cards)
	var positions: Dictionary = case_data.positions.duplicate(true)
	var challenge_errors: Array = []
	for id in case_data.get("reading_challenges", {}):
		var mode: String = case_data.reading_challenges[id].get("mode", "")
		if mode in ["misled", "lying"]:
			var source: String = card_by_id(view, id).get("source", "")
			var assumed: String = "good" if mode == "misled" else "evil"
			var allowed: Array = ["good", "marked"] if mode == "misled" else ["evil", "demon", "minion"]
			if positions.get(source, assumed) not in allowed:
				challenge_errors.append("The selected reader alignment conflicts with the %s explanation." % mode.capitalize())
			else:
				positions[source] = positions.get(source, assumed)
	var proposed = solve(view, proposed_cards, positions, case_data.answers, knowledge, case_data.conclusion.at, case_data.get("temporal_positions", []))
	var errors = validate_answers(view, case_data) + case_data.get("graph_errors", [])
	errors.append_array(challenge_errors)
	var conclusion: Dictionary = case_data.conclusion
	if conclusion.type in ["evil", "good", "marked"] and not proposed.worlds.is_empty() and not proposed.worlds.any(func(w): return holds(view, w, conclusion)):
		errors.append("The conclusion contradicts every world allowed by these premises and choices.")
	var n = 0
	var alternative = {}
	for world in base.worlds:
		if holds(view, world, conclusion):
			n += 1
		elif alternative.is_empty():
			alternative = describe_world(view, world, conclusion.at)
	var status = "incomplete"
	if not errors.is_empty() or proposed.worlds.is_empty() or not proposed.movement_ok:
		status = "conflicting"
	elif proposed.worlds.size() == 1:
		status = "coherent"
	var entailment = "possible"
	if base.worlds.is_empty() or not base.movement_ok:
		entailment = "conflicting"
	elif conclusion.type == "conflict":
		entailment = "unsupported"
	elif n == base.worlds.size():
		entailment = "forced"
	elif n == 0:
		entailment = "unsupported"
	if conclusion.type == "conflict" and (base.worlds.is_empty() or not base.movement_ok):
		entailment = "forced"
	var conflicts: Array = []

		conflicts = minimal_conflicts(view, cards, knowledge)
	return {"status": status, "entailment": entailment, "compatible": base.worlds.size(), "supporting": n, "hypothetically_released": cards.filter(func(c): return c not in proposed_cards).map(func(c): return c.id),

		"movement": proposed.movement, "movement_ok": proposed.movement_ok,
		"assumptions": assumption_labels(view, cards), "unresolved": unresolved(view, proposed.worlds), "conflicts": conflicts}

static func minimal_conflicts(view: Dictionary, cards: Array, knowledge: Dictionary = {}) -> Array:
	var inputs = [_query_view(view), cards, _query_knowledge(knowledge)]
	var key = hash(inputs)
	if reuse_solves and conflict_cache.has(key) and conflict_cache[key].inputs == inputs:
		return conflict_cache[key].result
	# Shrink one conflict, then explore excluding each member to discover other cores.
	# This avoids enumerating every unrelated subset for a simple alibi conflict.
	cards = cards.filter(func(c): return c.kind in ["report", "sighting", "alibi", "totem", "dropped"])
	var result: Array = []
	_find_conflicts(view, cards, knowledge, result, {})
	if reuse_solves:
		if conflict_cache.size() >= 128:
			conflict_cache.erase(conflict_cache.keys()[0])
		conflict_cache[key] = {"inputs": inputs.duplicate(true), "result": result}
	return result

static func _find_conflicts(view: Dictionary, cards: Array, knowledge: Dictionary, result: Array, visited: Dictionary) -> void:
	var ids: Array = cards.map(func(c): return c.id)
	var key = str(ids)
	if visited.has(key):
		return
	visited[key] = true
	var solved = solve(view, cards, {}, {}, knowledge)
	if not solved.worlds.is_empty() and solved.movement_ok:
		return
	var core = cards.duplicate()
	for card in cards:
		var trial = core.filter(func(c): return c.id != card.id)
		var tested = solve(view, trial, {}, {}, knowledge)
		if tested.worlds.is_empty() or not tested.movement_ok:
			core = trial
	var conflict: Array = core.map(func(c): return c.id)
	if not result.has(conflict):
		result.append(conflict)
	for card in core:
		_find_conflicts(view, cards.filter(func(c): return c.id != card.id), knowledge, result, visited)

static func solve(view: Dictionary, cards: Array, positions: Dictionary = {}, answers: Dictionary = {}, knowledge: Dictionary = {}, epoch: String = "", temporal_positions: Array = []) -> Dictionary:
	var span = Profiler.open("constraint_solve")
	# Bounded search, per-person theories and diagnostic explanations repeatedly ask
	# exactly the same question. Cache complete public inputs, including chronology
	# and private knowledge; never a partial key that could reuse an older night.
	var inputs = [_query_view(view), cards, positions, answers, _query_knowledge(knowledge), epoch, temporal_positions]
	var key = hash(inputs)
	if reuse_solves and solve_cache.has(key) and solve_cache[key].inputs == inputs:
		Profiler.count("solve_cache_hit")
		var cached: Dictionary = solve_cache[key].result
		Profiler.close(span)
		return cached
	var result = _solve(view, cards, positions, answers, knowledge, epoch, temporal_positions)
	if reuse_solves:
		if solve_cache.size() >= 1024:
			solve_cache.erase(solve_cache.keys()[0])
		solve_cache[key] = {"inputs": inputs.duplicate(true), "result": result}
	Profiler.close(span)
	return result

## The explicit cards argument already contains the selected receipts. Unselected
## library cards and attention bookkeeping are not inputs to _solve, roles or routes;
## retaining them in keys prevented identical scoped/union questions from sharing.
static func _query_view(view: Dictionary) -> Dictionary:
	var result = view.duplicate()
	result.erase("cards")
	return result

static func _query_knowledge(knowledge: Dictionary) -> Dictionary:
	var result = knowledge.duplicate()
	result.erase("own_reports")
	result.erase("attention_reports")
	return result

## Every legal world for this view (cached). Shared by the solve and by CaseEval.
static func universe(view: Dictionary) -> Array:
	return universes[_universe_key(view)] if universes.has(_universe_key(view)) else _build_universe(view)

static func _universe_key(view: Dictionary) -> String:
	return str([person_ids(view), view.evil_count, view.get("chronology", {}), view.get("world_options", {})])

static func _build_universe(view: Dictionary) -> Array:
	var ids: Array = person_ids(view)
	var key = _universe_key(view)
	if universes.size() >= 16:
		universes.clear()
	var all_worlds: Array = []
	if Domain.live(view):
		all_worlds = Domain.worlds(view)
		if view.get("chronology", {}).has("curse"):
			var expanded: Array = []
			for world in all_worlds:
				for candidate in TimeModel.expand(view, world):
					candidate["__i"] = expanded.size()
					expanded.append(candidate)
			all_worlds = expanded
	else:
		for evil in range(1 << ids.size()):
			if popcount(evil) == view.evil_count:
				for marked in range(ids.size()):
					if not evil & (1 << marked):
						all_worlds.append_array(TimeModel.expand(view, {"evil": evil, "marked": marked}))
	universes[key] = all_worlds
	return all_worlds

## For one report (or curse event) card: a byte per universe world, 1 where the card's
## content holds as a healthy reading from a good source (cached by exact inputs).
## marked: answered "registered because Marked" (Dowser and Surveyor answers only).
static func report_table(view: Dictionary, card: Dictionary, positive: Array = [], positive_blocks: Array = [], marked: Array = []) -> PackedByteArray:
	var universe_key = _universe_key(view)
	var universe: Array = universe(view)
	positive = positive.duplicate()
	positive.sort()
	positive_blocks = positive_blocks.duplicate()
	positive_blocks.sort()
	marked = marked.duplicate()
	marked.sort()
	# Homes affect Surveyor blocks; kind/event distinguish public process cards.
	# Hashes accelerate lookup, but exact snapshots guard collisions and edits.
	var inputs = [universe_key, view.people, view.get("timeline", {}), card.kind, card.get("event", ""), card.source,
		card.get("role", ""), card.observed_at, card.data, positive, positive_blocks]
	if not marked.is_empty():
		inputs.append(marked)
	if card.get("testimony_only", false):
		inputs.append([true, card.get("reported_at")])
	var key = hash(inputs)
	if not report_matches.has(key) or report_matches[key].inputs != inputs:
		if report_matches.size() > 512:
			report_matches.clear()
		var accepted = PackedByteArray()
		accepted.resize(universe.size())
		# Worlds differing only in the demon's identity share one answer.
		var groups = sig_groups(view)
		var rep: PackedInt32Array = groups.rep
		var of: PackedInt32Array = groups.of
		var answers = PackedByteArray()
		answers.resize(rep.size())
		if card.get("testimony_only", false) and card.kind == "report":
			for i in range(universe.size()):
				accepted[i] = 1 if source_valid(view, universe[i], card) and _testimony_content(view, universe[i], card) else 0
			report_matches[key] = {"inputs": inputs.duplicate(true), "accepted": accepted}
			return accepted
		if card.kind == "report":
			# A valid source (vectorised), then the content where it matters.
			var valid: PackedByteArray = valid_table(view, card)
			var trivial: bool = card.get("role", "") not in CONTENT_ROLES
			for g in range(rep.size()):
				answers[g] = 0 if valid[rep[g]] == 0 else (1 if trivial or _report_content(view, universe[rep[g]], card, {"positive": positive, "blocks": positive_blocks, "marked": marked}) else 0)
		else:
			for g in range(rep.size()):
				answers[g] = 1 if _report_holds(view, universe[rep[g]], card, {"positive": positive, "blocks": positive_blocks, "marked": marked}) else 0
		for i in range(universe.size()):
			accepted[i] = answers[of[i]]
		report_matches[key] = {"inputs": inputs.duplicate(true), "accepted": accepted}
	return report_matches[key].accepted

## A world's key without the demon's identity. Report contents and source validity
## never read it, so worlds differing only there share one answer. -1: no key.
static func content_sig(view: Dictionary, world: Dictionary) -> int:
	if world.has("converted"):
		return int(world.get("__i", -1)) # Source validity now depends on the turnee.
	if not world.has("__key"):
		return -1
	if world.has("demon"):
		return int(world.__key) / (view.people.size() + 1)
	return int(world.__key)

## Roles whose report content can fail in a world (the rest only need a valid source).
const CONTENT_ROLES = ["surveyor", "dowser", "rumormonger", "nightwatch", "warden"]

static var _bits: Dictionary = {}

## Per universe: each world's evil bits and Delusional holders' bits, and whether any
## world has a conversion history (fixture curse worlds).
static func world_bits(view: Dictionary) -> Dictionary:
	var key = _universe_key(view)
	if _bits.has(key):
		return _bits[key]
	if _bits.size() > 16:
		_bits.clear()
	var universe: Array = universe(view)
	var evil = PackedInt64Array()
	var deluded = PackedInt64Array()
	evil.resize(universe.size())
	deluded.resize(universe.size())
	var plain = true
	for i in range(universe.size()):
		var w: Dictionary = universe[i]
		evil[i] = int(w.evil)
		deluded[i] = int(w.get("holders", 0)) if w.get("modifier", "") == "delusional" else 0
		if w.has("converted"):
			plain = false
	_bits[key] = {"evil": evil, "deluded": deluded, "plain": plain}
	return _bits[key]

## source_valid for every world of the universe.
static func valid_table(view: Dictionary, card: Dictionary) -> PackedByteArray:
	var universe: Array = universe(view)
	var t = PackedByteArray()
	t.resize(universe.size())
	t.fill(0)
	if card.kind != "report":
		t.fill(1)
		return t
	if card.get("testimony_only", false):
		for i in range(universe.size()):
			t[i] = 1 if source_valid(view, universe[i], card) else 0
		return t
	var live: bool = Domain.live(view)
	# Gift timing and the source being alive don't vary by world.
	if live and (not Domain.gift_time(card) or not preload("res://workshop/live_history.gd").observer_alive(view, card.source, card.observed_at)):
		return t
	var bits = world_bits(view)
	var source: int = person_ids(view).find(card.source)
	if live and bits.plain:
		# No conversions: a good source that isn't Delusional.
		if source >= 0:
			var bit: int = 1 << source
			var evil: PackedInt64Array = bits.evil
			var deluded: PackedInt64Array = bits.deluded
			for i in range(universe.size()):
				t[i] = 0 if (evil[i] & bit) or (deluded[i] & bit) else 1
		return t
	var groups = sig_groups(view)
	var rep: PackedInt32Array = groups.rep
	var of: PackedInt32Array = groups.of
	var answers = PackedByteArray()
	answers.resize(rep.size())
	for g in range(rep.size()):
		answers[g] = 1 if source_valid(view, universe[rep[g]], card) else 0
	for i in range(universe.size()):
		t[i] = answers[of[i]]
	return t

static var _sig_groups: Dictionary = {}

## Per universe: worlds grouped by content_sig (dense ids, one representative each).
## Worlds without a key get a group of their own.
static func sig_groups(view: Dictionary) -> Dictionary:
	var key = _universe_key(view)
	if _sig_groups.has(key):
		return _sig_groups[key]
	if _sig_groups.size() > 16:
		_sig_groups.clear()
	var universe: Array = universe(view)
	var index = {}
	var rep = PackedInt32Array()
	var of = PackedInt32Array()
	of.resize(universe.size())
	for i in range(universe.size()):
		var sig: int = content_sig(view, universe[i])
		var g: int = index.get(sig, -1) if sig >= 0 else -1
		if g < 0:
			g = rep.size()
			rep.append(i)
			if sig >= 0:
				index[sig] = g
		of[i] = g
	_sig_groups[key] = {"rep": rep, "of": of}
	return _sig_groups[key]

## Whether a card is constrained by a report table (reports, and the curse event).
static func tabled(card: Dictionary) -> bool:
	return card.kind == "report" or (card.kind == "event" and card.get("event") == "curse")

## The source-side conditions of _report_holds: a good, healthy source able to read then.
## Where this fails, the report says nothing about the world (CaseEval: likelihood q).
static func source_valid(view: Dictionary, world: Dictionary, card: Dictionary) -> bool:
	if card.kind != "report":
		return true
	if card.get("testimony_only", false):
		return not TimeModel.evil_at(view, world, card.source, TimeModel.statement_time(card))
	if TimeModel.evil_at(view, world, card.source, card.observed_at):
		return false
	if Domain.live(view):
		if not Domain.healthy(view, world, card.source) or not Domain.gift_time(card):
			return false
		if not preload("res://workshop/live_history.gd").observer_alive(view, card.source, card.observed_at):
			return false
	return true

static func _solve(view: Dictionary, cards: Array, positions: Dictionary = {}, answers: Dictionary = {}, knowledge: Dictionary = {}, epoch: String = "", temporal_positions: Array = []) -> Dictionary:
	if not positions.is_empty() or not temporal_positions.is_empty():
		# Candidate theories differ only in position filters. Reports, roles and routes
		# do not read those filters, so compute their common surviving worlds once.
		var base = solve(view, cards, {}, answers, knowledge, epoch)
		return {"worlds": _positions(view, base.worlds, positions, epoch, temporal_positions),
			"movement": base.movement, "movement_ok": base.movement_ok}
	# Report masks use the universe's array index, including distinct conversion
	# histories. Keep the worlds themselves unchanged for explanations/checkpoints.
	var universe: Array = universe(view)
	var tables: Array = []
	for card in cards:
		if not tabled(card):
			continue
		tables.append(report_table(view, card, answers.get(card.id, {}).get("positive", []), answers.get(card.id, {}).get("blocks", []), answers.get(card.id, {}).get("marked", [])))
	var worlds: Array = []
	if tables.is_empty():
		worlds = universe.duplicate()
	else:
		for i in range(universe.size()):
			var accepted = true
			for table in tables:
				if table[i] == 0:
					accepted = false
					break
			if accepted:
				worlds.append(universe[i])
	if Domain.live(view) and not Domain.claims_compatible(cards, knowledge):
		worlds = []
	var has_curse: bool = view.get("chronology", {}).has("curse")
	if not has_curse and temporal_positions.is_empty():
		var must_be_good: int = 0
		for id in knowledge.get("good", []):
			var idx = person_index(view, id)
			if idx >= 0:
				must_be_good |= (1 << idx)
		var must_be_evil: int = 0
		var marked_req: int = -2
		var other_positions: Array = []
		for id in positions:
			var idx = person_index(view, id)
			if idx < 0:
				if positions[id] in ["evil", "marked", "unmarked"]:
					worlds = []
				continue
			var pos_val = positions[id]
			if pos_val == "good":
				must_be_good |= (1 << idx)
			elif pos_val == "evil":
				must_be_evil |= (1 << idx)
			elif pos_val == "marked":
				if marked_req >= 0 and marked_req != idx:
					worlds = [] # One world cannot have two different Marked people.
				marked_req = idx
			else:
				other_positions.append({"id": id, "pos": pos_val})
		if must_be_good != 0 or must_be_evil != 0 or marked_req != -2:
			worlds = worlds.filter(func(w):
				if (w.evil & must_be_good) != 0:
					return false
				if (w.evil & must_be_evil) != must_be_evil:
					return false
				if marked_req >= 0 and w.marked != marked_req:
					return false
				return true)
		for entry in other_positions:
			worlds = worlds.filter(func(w): return position_holds(view, w, entry.id, entry.pos, epoch))
	else:
		for id in knowledge.get("good", []):
			worlds = worlds.filter(func(w): return position_holds(view, w, id, "good", knowledge.get("good_epoch", epoch)))
		for id in positions:
			worlds = worlds.filter(func(w): return position_holds(view, w, id, positions[id], epoch))
		for assignment in temporal_positions:
			for id in assignment.positions:
				worlds = worlds.filter(func(w): return position_holds(view, w, id, assignment.positions[id], assignment.at))
	for fact in knowledge.get("timed_good", []):
		worlds = worlds.filter(func(w): return not TimeModel.evil_at(view, w, fact.person, fact.at))
	var movement_cards: Array = cards.filter(func(c): return c.kind in ["sighting", "alibi"] or (c.kind == "report" and c.get("role", "") in ["nightwatch", "lamplighter"]))
	if view.get("world_options", {}).get("roles", false):
		# One role table for the surviving worlds (same answers as fits per world).
		var fit: PackedByteArray = Roles.fits_table(view, worlds, cards, knowledge, answers)
		var kept: Array = []
		for i in range(worlds.size()):
			if fit[i] == 1:
				kept.append(worlds[i])
		worlds = kept
	var movement = movement_for(view, cards if view.get("live_activity", false) else movement_cards, answers, knowledge)
	return {"worlds": worlds, "movement": movement.nights, "movement_ok": movement.okay}

static func _positions(view: Dictionary, worlds: Array, positions: Dictionary, epoch: String, temporal: Array) -> Array:
	if view.get("chronology", {}).has("curse") or not temporal.is_empty():
		for id in positions:
			worlds = worlds.filter(func(w): return position_holds(view, w, id, positions[id], epoch))
		for assignment in temporal:
			for id in assignment.positions:
				worlds = worlds.filter(func(w): return position_holds(view, w, id, assignment.positions[id], assignment.at))
		return worlds
	var good = 0
	var evil = 0
	var marked = -2
	var other: Array = []
	for id in positions:
		var index = person_index(view, id)
		if index < 0:
			if positions[id] in ["evil", "marked", "unmarked"]:
				return []
			continue
		match positions[id]:
			"good": good |= 1 << index
			"evil": evil |= 1 << index
			"marked":
				if marked >= 0 and marked != index:
					return []
				marked = index
			_: other.append(id)
	worlds = worlds.filter(func(w): return not (w.evil & good) and (w.evil & evil) == evil and (marked < 0 or w.marked == marked))
	for id in other:
		worlds = worlds.filter(func(w): return position_holds(view, w, id, positions[id], epoch))
	return worlds

static func movement_for(view: Dictionary, cards: Array, answers: Dictionary = {}, knowledge: Dictionary = {}) -> Dictionary:
	if preload("res://workshop/live_activity.gd").available(view):
		return preload("res://workshop/live_activity.gd").routes(view, cards, answers, knowledge)
	var ids: Array = view.people.map(func(p): return p.id)
	var movement_cards: Array = cards.filter(func(c): return c.kind in ["sighting", "alibi"] or (c.kind == "report" and c.get("role", "") in ["nightwatch", "lamplighter"]))
	var movement_key = JSON.stringify([ids, movement_cards, answers, knowledge])
	if not movement_cache.has(movement_key):
		if movement_cache.size() > 256:
			movement_cache.clear()
		movement_cache[movement_key] = _movement(view, movement_cards, answers, knowledge)
	return movement_cache[movement_key]

static func popcount(bits: int) -> int:
	var count = 0
	while bits:
		bits &= bits - 1
		count += 1
	return count

static func position_holds(view: Dictionary, world: Dictionary, id: String, position: String, epoch: String = "") -> bool:
	if position in ["demon", "minion"]:
		var i: int = person_index(view, id)
		return i >= 0 and (world.get("demon", -1) == i if position == "demon" else ((world.evil & (1 << i)) != 0 and world.get("demon", -1) != i))
	if position in ["marked", "unmarked"]:
		var index: int = person_index(view, id)
		if index < 0:
			return false
		return world.marked == index if position == "marked" else world.marked != index
	var evil: bool = TimeModel.evil_at(view, world, id, view.alignment_epoch if epoch == "" else epoch)
	return evil if position == "evil" else not evil

static func _report_holds(view: Dictionary, world: Dictionary, card: Dictionary, answer: Dictionary) -> bool:
	_ensure_people(view)
	if card.kind == "event" and card.get("event") == "curse":
		return answer.get("positive", []).is_empty() or view.people[world.get("converted", -1)].id in answer.positive
	if card.kind != "report":
		return true
	if card.get("testimony_only", false):
		return source_valid(view, world, card) and _testimony_content(view, world, card)
	# An accepted healthy gift assumes a good source at the reading's time.
	return source_valid(view, world, card) and _report_content(view, world, card, answer)

## Cheap compatibility of the recounted content, with no anonymous role holder.
## This is weak testimony, not a faithful original reading or a role proof.
static func _testimony_content(view: Dictionary, world: Dictionary, card: Dictionary) -> bool:
	_ensure_people(view)
	var data: Dictionary = card.data
	var pair = 0
	for person in data.get("pair", []):
		pair |= 1 << person_index(view, person)
	var evil: int = world.evil
	if world.has("converted") and TimeModel.evil_index(view, world, world.converted, card.observed_at):
		evil |= 1 << int(world.converted)
	match card.role:
		"seer":
			var demon: bool = world.get("demon", -1) >= 0 and (pair & (1 << int(world.demon))) != 0
			return demon or (pair & ~evil) != 0 if data.yes else not demon
		"clerk": return (pair & ~world.evil) != 0
		"archivist": return (pair & world.evil & ~(1 << int(world.get("demon", -1)))) != 0
		"gravedigger":
			var target: int = person_index(view, data.executed)
			var reads_evil: bool = (world.evil & (1 << target)) != 0 or world.marked == target
			return reads_evil if Domain.C.ROLES.get(data.role, {}).get("align") == "evil" else (world.evil & (1 << target)) == 0
		"rumormonger": return world.get("modifier", "marked") == data.secret and (int(world.get("holders", 1 << world.marked)) & pair) != 0
	return _report_content(view, world, card, {})

## What a report says, given a valid source (see source_valid): does it hold here?
static func _report_content(view: Dictionary, world: Dictionary, card: Dictionary, answer: Dictionary) -> bool:
	_ensure_people(view)
	if card.role == "warden":
		return not view.get("timeline", {}).get("events", []).any(func(event): return event.kind == "demon_death" and event.at.day == card.observed_at.day and card.data.target in event.subjects)
	if Domain.live(view):
		if card.role == "nightwatch" and card.data.seen.any(func(id): return not preload("res://workshop/live_history.gd").alive_at_night(view, id, card.night)):
			return false
		if card.role == "rumormonger":
			return Domain.rumor(view, world, card, answer)
	var d: Dictionary = card.data
	if card.role == "rumormonger":
		var marked: int = world.marked
		var marked_id: String = _cached_ids[marked] if marked >= 0 and marked < _cached_ids.size() else ""
		return marked_id in d.pair and (answer.get("positive", []).is_empty() or marked_id in answer.positive)
	if card.role not in ["surveyor", "dowser"]:
		return true
	# A Marked person may register either way, independently for each reading.
	var evil_bits: int = world.evil
	if world.has("converted") and world.converted >= 0:
		if TimeModel.evil_index(view, world, world.converted, card.observed_at):
			evil_bits |= (1 << world.converted)
	var ans_pos = answer.get("positive", [])
	var ans_mask: int = 0
	for id in ans_pos:
		var index = person_index(view, id)
		if index < 0:
			return false
		ans_mask |= 1 << index
	var group_mask: int = 0
	if card.role == "dowser":
		for id in d.group:
			group_mask |= (1 << person_index(view, id))
	var is_surveyor: bool = card.role == "surveyor"
	var target_count: int = d.count
	_ensure_people(view)
	var b_masks: Array = _cached_block_masks.values()
	var chosen_blocks: Array = answer.get("blocks", [])
	if not is_surveyor and not chosen_blocks.is_empty():
		return false
	for block in chosen_blocks:
		if not _cached_block_masks.has(block):
			return false
	# "Registered because Marked": that person is the Marked one and read as evil.
	var required_marked: Array = answer.get("marked", [])
	for id in required_marked:
		if person_index(view, id) != world.marked:
			return false
	for mark_reads in ([true] if not required_marked.is_empty() else [false, true]):
		var pos_mask: int = evil_bits | ((1 << world.marked) if (mark_reads and world.marked >= 0) else 0)
		if (pos_mask & ans_mask) != ans_mask:
			continue
		if not chosen_blocks.all(func(block): return bool(pos_mask & _cached_block_masks[block])):
			continue
		var count: int = 0
		if is_surveyor:
			for b in b_masks:
				if (pos_mask & b) != 0:
					count += 1
		else:
			count = popcount(pos_mask & group_mask)
		if count == target_count:
			return true
	return false

static func _movement(view: Dictionary, cards: Array, answers: Dictionary, knowledge: Dictionary = {}) -> Dictionary:
	var ids: Array = view.people.map(func(p): return p.id)
	var nights = {}
	var okay = true
	var observed_nights: Array = []
	for card in cards:
		if card.get("night", -1) >= 0 and card.night not in observed_nights:
			observed_nights.append(card.night)
	for night in observed_nights:
		var relevant: Array = cards.filter(func(c): return not c.get("quoted", false) and c.get("night", -1) == night and (c.kind in ["sighting", "alibi"] or c.get("role", "") in ["nightwatch", "lamplighter"]))
		if relevant.is_empty():
			continue
		var possibilities: Array = []
		for mask in range(1 << ids.size()):
			var valid = true
			var own: String = knowledge.get("self", "")
			if own in ids and knowledge.get("own_outings", {}).has(str(night)):
				if ((mask & (1 << ids.find(own))) != 0) != knowledge.own_outings[str(night)]:
					continue
			for card in relevant:
				var required: Array = []
				if card.kind == "sighting":
					required = card.get("subjects", [card.subject])
				elif card.kind == "alibi":
					if mask & (1 << ids.find(card.subject)):
						valid = false
				elif card.role == "nightwatch":
					required = card.data.seen + [card.source]
				elif card.role == "lamplighter":
					required = [card.source] + answers.get(card.id, {}).get("positive", [])
					if popcount(mask & ~(1 << ids.find(card.source))) != card.data.count:
						valid = false
				for id in required:
					if not (mask & (1 << ids.find(id))):
						valid = false
				if not valid:
					break
			if valid:
				possibilities.append(mask)
		var forced: Array = []
		for i in range(ids.size()):
			if not possibilities.is_empty() and possibilities.all(func(mask): return (mask & (1 << i)) != 0):
				forced.append(ids[i])
		nights[str(night)] = {"compatible": possibilities.size(), "forced_out": forced}
		okay = okay and not possibilities.is_empty()
	return {"okay": okay, "nights": nights}

static func holds(view: Dictionary, world: Dictionary, conclusion: Dictionary) -> bool:
	var index: int = person_index(view, conclusion.target)
	if index < 0:
		return false
	match conclusion.type:
		"demon", "minion": return position_holds(view, world, conclusion.target, conclusion.type, conclusion.at)
		"notdemon": return not position_holds(view, world, conclusion.target, "demon", conclusion.at)
		"evil", "suspicious": return TimeModel.evil_index(view, world, index, conclusion.at)
		"good": return not TimeModel.evil_index(view, world, index, conclusion.at)
		"marked": return world.marked == index
		"secret_mod": return world.get("modifier", "marked") == conclusion.get("mod", "") and (int(world.get("holders", 1 << world.marked)) & (1 << index)) != 0
		"role", "notrole":
			# A declaration question, not an extra hidden role assignment: the live
			# adapter requires a cited matching claim. Good declarations are binding;
			# the existing joint role table weighs whether those claims fit together.
			var claimed: bool = not TimeModel.evil_index(view, world, index, conclusion.at)
			return claimed if conclusion.type == "role" else not claimed
		"lied", "truthful":
			# Existing challenge semantics: evil-reader versus honest-reader
			# explanations. Impairment is not deliberate dishonesty.
			var evil: bool = TimeModel.evil_index(view, world, index, conclusion.get("question_at", conclusion.at))
			return evil if conclusion.type == "lied" else not evil
		"misled", "unreliable":
			var table: PackedByteArray = question_table(view, conclusion)
			var i: int = world.get("__i", -1)
			if i < 0:
				i = universe(view).find(world)
			return i >= 0 and i < table.size() and table[i] != 0
	return false

static func question_table(view: Dictionary, conclusion: Dictionary) -> PackedByteArray:

static func describe_world(view: Dictionary, world: Dictionary, epoch: String = "") -> Dictionary:

static func unresolved(view: Dictionary, worlds: Array) -> Array:
	var result: Array = []
	if worlds.is_empty():
		return result
	# A person is unresolved when they are evil in some worlds and good in others.
	var all_mask: int = (1 << view.people.size()) - 1
	var any_evil: int = 0
	var any_good: int = 0
	for w in worlds:
		any_evil |= w.evil
		any_good |= ~w.evil & all_mask
		if any_evil & any_good == all_mask:
			break
	for i in range(view.people.size()):
		if any_evil & any_good & (1 << i):
			result.append(view.people[i].id)
	return result

static func validate_answers(view: Dictionary, case_data: Dictionary) -> Array:
	var errors: Array = []
	for id in case_data.answers:
		if id not in case_data.cards:
			continue
		var card = card_by_id(view, id)
		var positive: Array = case_data.answers[id].get("positive", [])
		var chosen_blocks: Array = case_data.answers[id].get("blocks", [])
		var maximum: int = (0 if card.data.get("yes") == false else 1) if card.get("role") == "seer" else (1 if card.get("role", "") in ["rumormonger", "clerk", "archivist"] else card.get("data", {}).get("count", 10))
		if positive.size() + chosen_blocks.size() > maximum:
			errors.append("%s: there are more answers than positive slots." % id)
		var unique: Array = []
		for person in positive:
			if person not in view.people.map(func(p): return p.id):
				errors.append("%s: choose a known person." % id)
				continue
			if person in unique:
				errors.append("%s: the same person fills two slots." % id)
			unique.append(person)
		if card.get("role", "") == "dowser" and not positive.all(func(p): return p in card.data.group):
			errors.append("%s: choose people in this reading's group." % id)
		if card.get("role", "") in ["clerk", "archivist", "seer"] and not positive.all(func(p): return p in card.data.pair):
			errors.append("%s: choose a person in this reading's pair." % id)
		if card.get("role") == "seer" and case_data.answers[id].get("registration", "") not in ["", "demon", "marked", "red_herring"]:
			errors.append("%s: choose a valid registration explanation." % id)
		if card.get("role", "") == "lamplighter" and card.source in positive:
			errors.append("%s: the count excludes the Lamplighter." % id)
		if card.get("role", "") == "surveyor":
			var blocks: Array = []
			for block in chosen_blocks:
				if block not in C.BLOCKS:
					errors.append("%s: choose a known block." % id)
				if block in blocks:
					errors.append("%s: each slot needs a different block." % id)
				blocks.append(block)
			for person in positive:
				var matches: Array = view.people.filter(func(p): return p.id == person)
				if matches.is_empty():
					continue
				var home: String = matches[0].home
				if home in blocks:
					errors.append("%s: each slot needs a different block." % id)
				blocks.append(home)
		elif not chosen_blocks.is_empty():
			errors.append("%s: only Surveyor answers can name blocks." % id)
		if card.get("role", "") == "rumormonger" and not positive.all(func(p): return p in card.data.pair):
			errors.append("%s: choose someone in the named pair." % id)
	return errors

static func person_name(view: Dictionary, id: String) -> String:
	for p in view.people:
		if p.id == id:
			return p.name
	return id

## Only interacting role/route predicates share factors. There is no enumeration
## of the Cartesian product of source states, and no source is dropped to repair

static var cache = {}

static func assess(view: Dictionary, cards: Array, knowledge: Dictionary, rates: Dictionary, q: Dictionary, baseline_ids: Array = [], representatives: PackedInt32Array = PackedInt32Array(), multiplicity: PackedFloat64Array = PackedFloat64Array(), include_public: bool = true) -> Dictionary:
	var start = Time.get_ticks_usec()
	var input = [Solver._query_view(view), cards, Solver._query_knowledge(knowledge), include_public]
	var key = hash(input)
	if not cache.has(key) or cache[key].input != input:
		if cache.size() >= 32:
			cache.clear()
		cache[key] = {"input": input.duplicate(true), "graph": _compile(view, cards, knowledge, include_public)}
	var graph: Dictionary = cache[key].graph
	var compile_us = Time.get_ticks_usec() - start
	var value_us = 0
	var sum_us = 0
	var result = {"base": PackedFloat64Array(), "post": PackedFloat64Array(), "held": {}, "acceptance": {}, "consistent": false,
		"issues": graph.issues, "states": 0, "peak_rows": graph.get("program", {}).get("peak", 0)}
	if not graph.issues.is_empty():
		return result
	var universe: Array = Solver.universe(view)
	var size: int = universe.size() if representatives.is_empty() else representatives.size()
	result.post.resize(size)
	for id in graph.markers:
		var held = PackedFloat64Array()
		held.resize(size)
		result.held[id] = held
	var memo = {}
	for slot in range(size):
		var index: int = slot if representatives.is_empty() else representatives[slot]
		var world: Dictionary = universe[index]
		var value_start = Time.get_ticks_usec()
		var values: Array = []
		for factor in graph.factors:
			values.append(_values(view, graph, factor, world, index, rates, q))
		# Many modifier worlds have identical factors. Memoize the numerical program,
		# not an assumed role/alignment equivalence that could miss impairment.
		var signature = hash(values)
		value_us += Time.get_ticks_usec() - value_start
		var sum_start = Time.get_ticks_usec()
		if not memo.has(signature) or memo[signature].values != values:
			memo[signature] = {"values": values, "result": Sum.run(graph.circuit, values)}
		var summed: Dictionary = memo[signature].result
		sum_us += Time.get_ticks_usec() - sum_start
		result.post[slot] = summed.total
		for id in graph.markers:
			var marker: int = graph.markers[id]
			result.held[id][slot] = summed.adjoints[marker][0 if graph.factors[marker].scope.is_empty() else 1]
		# Consistency asks whether all original statements can hold together, not
		# whether some fallible explanation gives the evidence positive likelihood.
		if not result.consistent and summed.total > 0.0:
			result.consistent = _all_hold(view, graph, world, index)
	result.states = memo.size()
	result["timing_us"] = {"compile": compile_us, "values": value_us, "sum": sum_us}
	var total = 0.0
	for slot in range(size):
		total += result.post[slot] * (1.0 if multiplicity.is_empty() else multiplicity[slot])
	for id in graph.markers:
		var mass = 0.0
		for slot in range(size):
			mass += result.held[id][slot] * (1.0 if multiplicity.is_empty() else multiplicity[slot])
		result.acceptance[id] = mass / total if total > 0.0 else 0.0
	if not include_public:
		result.base = result.post.duplicate()
	else:
		var prior = assess(view, cards.filter(func(card): return card.id in baseline_ids), knowledge, rates, q, [], representatives, multiplicity, false)
		result.base = prior.post
	return result

static func _compile(view: Dictionary, cards: Array, knowledge: Dictionary, include_public: bool) -> Dictionary:
	var plan = Channels.compile(view, cards)
	var graph = {"issues": plan.issues, "channels": plan.channels, "next": plan.channels.size(), "factors": [], "gates": {}, "facts": [], "originals": {}, "markers": {}, "knowledge": knowledge}
	if not graph.issues.is_empty():
		return graph
	for i in range(plan.channels.size()):
		_add(graph, [i], {"kind": "prior", "channel": plan.channels[i]})
	var bodies = {}
	for statement in plan.statements:
		var card: Dictionary = statement.card
		if card.get("quoted", false):
			_fact(bodies, card, -1)
			continue
		var content = _and(graph, statement.content)
		graph.markers[card.id] = _add(graph, [content] if content >= 0 else [], {"kind": "static", "values": PackedFloat64Array([1, 1]) if content >= 0 else PackedFloat64Array([1]), "query": true})
		_fact(bodies, statement.body, content)
		if card.kind == "report":
			_claim(bodies, graph, card.source, card.role, card.observed_at, statement.identity)
			if not graph.originals.has(statement.origin):
				graph.originals[statement.origin] = {"routes": [], "card": card}
			graph.originals[statement.origin].routes.append(content)
		if card.has("relay_source"):
			_claim(bodies, graph, card.relay_source, "medium", card.relay_observed_at, statement.relay_identity)
			_fact(bodies, {"kind": "context", "source": card.source, "relay_source": card.relay_source, "relay_observed_at": card.relay_observed_at}, _and(graph, statement.relay_content))
	for body in bodies.values():
		var card: Dictionary = body.card
		card["id"] = "predicate:" + str(graph.facts.size())
		card["fact_id"] = card.id
		graph.facts.append({"card": card, "variable": _or(graph, body.routes)})
	for origin in graph.originals:
		var item: Dictionary = graph.originals[origin]
		var variable = _or(graph, item.routes)
		_add(graph, [variable] if variable >= 0 else [], {"kind": "chance", "origin": origin, "card": item.card})
	for fact in graph.facts:
		if Solver.tabled(fact.card):
			_add(graph, _scope([fact]), {"kind": "report", "fact": fact, "table": Solver.report_table(view, fact.card)})
	_movement(view, graph)
	_roles(view, graph)
	var allowed = PackedByteArray()
	allowed.resize(Solver.universe(view).size())
	for world in Solver.solve(view, [], {}, {}, knowledge).worlds:
		allowed[world.__i if world.has("__i") else Solver.universe(view).find(world)] = 1
	_add(graph, [], {"kind": "knowledge", "table": allowed})
	var public_factors: Array = Public.factors(view)
	if include_public and not public_factors.is_empty():
		_add(graph, [], {"kind": "public", "factors": public_factors})
	var speech: Array = graph.facts.filter(func(f): return f.card.kind in ["gift_hint", "refusal"])
	if not speech.is_empty():
		_conditional(graph, speech, "speech")
	graph.program = Sum.compile(graph.factors.map(func(factor): return factor.scope))
	graph.circuit = Sum.circuit(graph.program, graph.factors.map(func(f): return f.values if f.kind == "static" and not f.get("query", false) else null))
	return graph

static func _add(graph: Dictionary, scope: Array, data: Dictionary) -> int:
	data["scope"] = scope
	graph.factors.append(data)
	return graph.factors.size() - 1

static func _and(graph: Dictionary, variables: Array) -> int:
	var unique: Array = []
	for variable in variables:
		if variable == -2:
			return -2
		if variable >= 0 and variable not in unique:
			unique.append(variable)
	if unique.is_empty():
		return -1
	var output: int = unique[0]
	for i in range(1, unique.size()):
		output = _gate(graph, output, unique[i], false)
	return output

static func _or(graph: Dictionary, variables: Array) -> int:
	if -1 in variables:
		return -1
	var unique: Array = []
	for variable in variables:
		if variable >= 0 and variable not in unique:
			unique.append(variable)
	if unique.is_empty():
		return -2
	var output: int = unique[0]
	for i in range(1, unique.size()):
		output = _gate(graph, output, unique[i], true)
	return output

static func _gate(graph: Dictionary, a: int, b: int, either: bool) -> int:
	var key = str([mini(a, b), maxi(a, b), either])
	if graph.gates.has(key):
		return graph.gates[key]
	var output: int = graph.next
	graph.next += 1
	var values = PackedFloat64Array()
	values.resize(8)
	for row in range(8):
		var holds: bool = (row & 1 != 0 or row & 2 != 0) if either else (row & 3 == 3)
		values[row] = 1.0 if holds == (row & 4 != 0) else 0.0
	_add(graph, [a, b, output], {"kind": "static", "values": values})
	graph.gates[key] = output
	return output

static func _conditional(graph: Dictionary, facts: Array, kind: String, extra: Dictionary = {}) -> Dictionary:

## Each role-completion component only touches predicates that can constrain the
## same gift or the same holder. Hall matching stays exact inside that component;
## separate gifts/sources integrate independently.
static func _roles(view: Dictionary, graph: Dictionary) -> void:
	var facts: Array = graph.facts.filter(func(f): return _role_keys(f.card).size() > 0)
	var components: Array = []
	for fact in facts:
		var keys = _role_keys(fact.card)
		var combined: Array = [fact]
		var merged = true
		while merged:
			merged = false
			for i in range(components.size() - 1, -1, -1):
				if components[i].keys.any(func(k): return k in keys):
					combined.append_array(components[i].facts)
					for k in components[i].keys:
						if k not in keys:
							keys.append(k)
					components.remove_at(i)
					merged = true
		components.append({"facts": combined, "keys": keys})
	for component in components:
		_conditional(graph, component.facts, "roles")

	if card.kind == "report" and card.role in ["nightwatch", "lamplighter", "medium"]:

	# projection sees this same snapshot, including source credibility and priors.

	baseline.ledger = {}
	var primary_listener = baseline.duplicate(true)
	var modern: bool = engine == "case_eval"
	var result = _assess_target_eval(view, case_data, primary_listener, context) if modern else _assess_target(view, case_data, primary_listener, context)
	var signature_case = Graph.compile_case(view, case_data)
	signature_case.conclusion = {"type": "evil", "target": "all", "at": case_data.conclusion.at}
	var common_key = JSON.stringify(_canonical([_signature(view, signature_case), baseline.suspicion, baseline.npc_trust,
		baseline.knowledge, baseline.reading_reliability, baseline.trust, baseline.confidence,
		baseline.get("confidence_mode"), baseline.get("reasoning_mode"), baseline.get("insight")]))
	# context.headline_only: a scoped reply whose effects are never applied (the shared
	# union owns belief changes). Only the headline target's effect is computed; every
	# other person's projection, theory and CaseEval re-run is skipped.
	var only: Array = [case_data.conclusion.target] if context.get("headline_only", false) and view.people.any(func(p): return p.id == case_data.conclusion.target) else []
	var consequences = _consequences_eval(view, case_data, baseline, result, common_key, only) if modern else Consequences.assess(view, case_data, baseline, _pattern, common_key, result.theory)
	var effects = {}
	for person in view.people:
		var id: String = person.id
		if not consequences.effects.has(id):
			continue
		var entry: Dictionary = consequences.effects[id].duplicate(true)
		entry["baseline"] = baseline.suspicion.get(id, 0.1)
		entry["before"] = snappedf(before.get(id, 0.1), 0.000000001)
		entry["after"] = snappedf(clampf(entry.baseline + entry.requested, 0.0, 1.5), 0.000000001)
		entry["applied"] = snappedf(entry.after - entry.baseline, 0.000000001)
		entry["delta"] = snappedf(entry.after - entry.before, 0.000000001)
		entry["previous"] = old_effects.get(id, {}).get("applied", 0.0)
		effects[id] = entry
		listener.suspicion[id] = entry.after
	var target: String = case_data.conclusion.target
	listener.ledger[case_data.id] = primary_listener.ledger[case_data.id].duplicate(true)
	listener.ledger[case_data.id].effects = effects.duplicate(true)
	listener.ledger[case_data.id].applied = effects[target].applied
	listener.case_memory = primary_listener.case_memory.duplicate(true)
	listener.theories = consequences.theories
	# The headline's original confidence decision remains the reported decision.
	var theory_key: String = target + "|" + case_data.conclusion.at
	if case_data.conclusion.type in ["marked", "demon", "minion"]:
		theory_key += "|" + case_data.conclusion.type
	if primary_listener.theories.has(theory_key):
		listener.theories[theory_key] = primary_listener.theories[theory_key].duplicate(true)
	result["effects"] = effects
	var challenges = Support.ReadingChallenges.assess(view, case_data, baseline, Solver.selected(view, signature_case).filter(func(c): return c.id in result.reasoning.considered and c.id not in result.reasoning.private_checks.rejected))
	result["reading_challenges"] = challenges.results
	for challenge in challenges.results:
		result.response += " " + challenge.note
	result["claims"] = consequences.claims
	result["effect_rule"] = consequences.rule
	result.before_suspicion = effects[target].before
	result.after_suspicion = effects[target].after
	result.suspicion_delta = effects[target].delta
	result.calculation.old_case_effect = effects[target].previous
	result.calculation.new_case_effect = effects[target].applied
	result.calculation.rule = consequences.rule
	# Speak about accepted intermediate inferences separately from the headline.
	if not result.repeated:
		if effects[target].get("protection_signal", 0.0) < 0 and case_data.conclusion.type != "conflict":
			result.response += " The quiet night gives that protection claim some support, though other explanations remain."
		for claim in consequences.claims:
			if claim.conclusion.target != target and claim.conclusion.type == "evil" and claim.status == "entailed_in_attention" and claim.forced_mass > 0:
				result.response += " The reports I granted implicate %s, if sound. That conclusion does not by itself settle %s's alignment." % [Solver.person_name(view, claim.conclusion.target), Solver.person_name(view, target)]
				break
	return result

static func applied_to(entry: Dictionary, person: String) -> float:
	if entry.has("effects"):
		return float(entry.effects.get(person, {}).get("applied", 0.0))
	return float(entry.get("applied", 0.0)) if entry.get("target") == person else 0.0

static func _assess_target(view: Dictionary, case_data: Dictionary, listener: Dictionary, context: Dictionary = {}) -> Dictionary:
	var span = Profiler.open("headline")
	var result = _assess_target_inner(view, case_data, listener, context)
	Profiler.close(span)
	return result

static func _assess_target_inner(view: Dictionary, case_data: Dictionary, listener: Dictionary, context: Dictionary = {}) -> Dictionary:
	var original = case_data.duplicate(true)
	case_data = Graph.compile_case(view, case_data)
	var conclusion: Dictionary = case_data.conclusion
	var target: String = conclusion.target
	var before_suspicion: float = listener.suspicion.get(target, 0.1)
	var before_trust: float = listener.trust
	var old: Dictionary = listener.ledger.get(case_data.id, {})
	# Assess from the baseline without this case's own previous persuasion. Otherwise
	# accusing a witness would lower their credibility on every repetition.
	if not old.is_empty():
		listener.suspicion[old.target] = snappedf(listener.suspicion.get(old.target, 0.1) - old.applied, 0.000000001)
	var previous: Dictionary = listener.case_memory.get(case_data.id, {})
	var signature = _signature(view, case_data)
	var input_key = JSON.stringify(_canonical([signature, listener.get("reasoning_mode", "full"), listener.get("insight", 0.5),
		listener.get("confidence_mode", "scale"), listener.confidence, listener.trust, listener.npc_trust,
		listener.suspicion, listener.reading_reliability, listener.knowledge]))
	var repeated: bool = previous.get("input_key", "") == input_key
	var knowledge: Dictionary = listener.knowledge
	var full_case = case_data.duplicate(true)
	for id in knowledge.get("own_reports", []):
		if id not in full_case.cards:
			full_case.cards.append(id)
	var assessment = Solver.analyze(view, full_case, knowledge)
	var shared = Solver.analyze(view, case_data)
	var overall = Solver.solve(view, Solver.selected(view, full_case), {}, {}, knowledge)
	var logical_cards: Array = Solver.selected(view, case_data).filter(func(c): return c.kind == "report" and c.role in ["surveyor", "rumormonger", "dowser"])
	var best = {"weight": 0.0, "agreement": 0.0, "quality": 0.0, "cards": [], "kind": "none"}
	# Consider independent support paths. Adding an optional premise cannot erase an
	# already valid smaller path; a contradiction in the asserted case can reopen it.
	# The legacy diagnostic enumerates every report subset. Live comparisons already
	# retain the real Argue result; avoid computing this superseded experiment too.
	var diagnostic_paths: int = 1 if Solver.Domain.live(view) else 1 << logical_cards.size()
	for mask in range(1, diagnostic_paths):
		var path: Array = []
		for i in range(logical_cards.size()):
			if mask & (1 << i):
				path.append(logical_cards[i].id)
		var subcase = Solver.empty_case()
		subcase.cards = path.duplicate()
		subcase.conclusion = conclusion
		for id in knowledge.get("own_reports", []):
			if id not in subcase.cards:
				subcase.cards.append(id)
		var analysis = Solver.analyze(view, subcase, knowledge)
		var agreement = 1.0 if analysis.entailment == "forced" else 0.0
		if analysis.entailment == "possible":
			agreement = _possibility_agreement(view, overall.worlds, conclusion, listener)
		if conclusion.type == "conflict":
			agreement = 0.0 # handled using the whole asserted set below
		var quality = _quality(view, path, listener)
		var weight = agreement * quality
		if weight > best.weight:
			best = {"weight": weight, "agreement": agreement, "quality": quality, "cards": path, "kind": "reading"}
		if shared.entailment == "possible" and conclusion.type != "conflict":
			var positions: Dictionary = case_data.positions.duplicate(true)
			positions.erase(target)
			var answers: Dictionary = case_data.answers.duplicate(true)
			if conclusion.type == "marked":
				for id in answers.keys():
					if Solver.card_by_id(view, id).get("role", "") == "rumormonger":
						answers.erase(id)
			var conditional_path = Solver.solve(view, Solver.selected(view, subcase), positions, answers, knowledge)
			if not conditional_path.worlds.is_empty() and conditional_path.worlds.all(func(w): return Solver.holds(view, w, conclusion)):
				if 0.4 * quality > best.weight:
					best = {"weight": 0.4 * quality, "agreement": 0.4, "quality": quality, "cards": path, "kind": "conditional"}
	var pattern = _pattern(view, case_data, listener)
	var legacy = best.duplicate(true)
	if Solver.Domain.live(view):
		legacy["omitted"] = true
	var reasoning = Support.explain(view, original, listener)
	if listener.get("reasoning_mode", "full") == "bounded":
		var attended: Dictionary = Reasoner.prepare(view, original, listener).case
		attended.cards = attended.cards.filter(func(id): return id not in reasoning.private_checks.rejected)
		pattern = _pattern(view, attended, listener)
		var considered_facts: Array = reasoning.considered.map(func(id): return Solver.semantic_key(Solver.card_by_id(view, id)))
		considered_facts.sort()
		input_key += JSON.stringify([considered_facts, reasoning.connections])
		repeated = previous.get("input_key", "") == input_key
	var theory = Theory.prepare(view, case_data, listener, reasoning, input_key)
	var combined = Support.assess(view, original, listener, theory.bias)
	var paths: Array = []
	for path in combined.paths:
		for id in path:
			if id not in paths:
				paths.append(id)
	if paths.is_empty():
		paths = logical_cards.map(func(c): return c.id)
	best = {"weight": combined.agreement, "agreement": combined.agreement, "quality": 1.0,
		"cards": paths, "kind": "conditional" if combined.conditional_mass > combined.forced_mass else "reading"}
	if conclusion.type == "suspicious" and pattern.weight > best.weight:
		best = pattern
	if conclusion.type == "suspicious" and Movement.available(view) and combined.evidence_agreement > 0:
		best.agreement = maxf(best.agreement, 0.5)
		best.weight = maxf(best.weight, 0.5)
	var conflict = assessment.status == "conflicting"
	if conclusion.type == "conflict":
		var detected: bool = (reasoning.status == "inconsistent" or not reasoning.private_checks.rejected.is_empty()) if listener.get("reasoning_mode", "full") == "bounded" else assessment.entailment == "forced"
		var agreement = 1.0 if detected else 0.0
		best = {"weight": agreement * _quality(view, case_data.cards, listener), "agreement": agreement, "quality": _quality(view, case_data.cards, listener), "cards": case_data.cards, "kind": "conflict"}
	elif listener.get("reasoning_mode", "full") != "bounded" and (not assessment.errors.is_empty() or (assessment.proposed == 0 and assessment.compatible > 0)):
		best.weight = 0.0
		best.agreement = 0.0
	if reasoning.private_checks.fatal:
		best.weight = 0.0
		best.agreement = 0.0
	var player_factor: float = 0.5 + listener.trust
	var confidence_factor: float = 1.0 if listener.get("confidence_mode", "scale") == "persistence" else 1.4 - 0.8 * listener.confidence
	# Venue is deliberately the same: no private/public/nomination duplicate effects.
	var strength: float = best.weight * player_factor * confidence_factor
	var coefficient: float = {"evil": 0.45, "demon": 0.45, "minion": 0.45, "good": -0.4, "marked": -0.1, "suspicious": 0.3, "conflict": 0.0}.get(conclusion.type, 0.0)
	var effect: float = strength * coefficient
	# Replace this case's complete prior effect, even if target/conclusion changed.
	var baseline: float = listener.suspicion.get(target, 0.1)
	listener.suspicion[target] = snappedf(clampf(baseline + effect, 0.0, 1.5), 0.000000001)
	var applied: float = snappedf(listener.suspicion[target] - baseline, 0.000000001)
	listener.ledger[case_data.id] = {"target": target, "applied": applied, "cards": best.cards.duplicate(), "version": case_data.version,
		"conclusion": conclusion.duplicate(true), "signature": signature, "report_acceptance": combined.report_acceptance.duplicate(true)}
	# No recurring trust reward for restating a case. Workshop trust is a controlled input.
	var changes = _changes(view, previous.get("case", {}), case_data)
	listener.case_memory[case_data.id] = {"signature": signature, "input_key": input_key, "version": case_data.version, "cards": case_data.cards.duplicate(), "strength": strength, "case": case_data.duplicate(true)}
	var objection = _objection(view, case_data, listener, shared, assessment, best, pattern)
	var stance = "accept" if best.agreement >= 0.6 else ("partial" if best.agreement >= 0.35 else "reject")
	var delta: float = listener.suspicion[target] - before_suspicion
	var response = ""
	if repeated:
		response = "That is the same case. Repeating it adds no new reason."
	elif conclusion.type == "conflict":
		response = "Those accounts cannot all be accurate. I have reopened which report to believe; this does not establish anyone's alignment." if best.agreement > 0 else "I don't see a conflict among the reports you selected."
	elif best.kind == "pattern" and best.weight > 0:
		response = "That gives me a reason to be suspicious, though it does not establish evil."
	elif conflict and conclusion.type != "conflict":
		response = "I cannot accept all those premises together. I give weight only to the explanations that survive doubting a report." if best.weight > 0 else "Your selected explanation does not fit. The underlying reports can still change my assessments."
	elif best.kind == "conditional":
		response = "That follows if I grant your extra choices. You still need to establish those choices."
	elif stance == "accept" and Movement.available(view) and assessment.entailment == "possible":
		response = "These observations make your explanation convincing to me, though an alternative remains logically possible."
	elif stance == "accept":
		response = "The conclusion follows if those reports are sound. My doubt is in the premises."
	elif combined.agreement < 0:
		response = "The supplied evidence weighs against that conclusion."
	elif assessment.entailment == "forced":
		response = "The inference follows if the reports are accurate. I give it some weight, but I still doubt the supporting reports."
	else:
		response = "I can see that possibility, but the reports also allow another explanation."
	if not combined.factors.is_empty():
		if combined.factors.any(func(label): return label.begins_with("Movement / trace")):
			response += " The routes, observation coverage and traces change which explanation I favor. Lawful activity and eligible passers remain alternatives; this is circumstantial evidence."
		else:
			response += " If the linked observation is accurate, the home alibi is false. That is circumstantial evidence, not proof of deliberate deception or evil." if combined.factors.any(func(label): return label.begins_with("Challenged home alibi")) else " The supplied test factor changes which remaining explanation I favor; it does not prove it."
	if listener.get("reasoning_mode", "full") == "bounded":
		objection = _bounded_objection(reasoning)
		if reasoning.private_checks.fatal:
			response = reasoning.private_checks.objections[0]
		elif conclusion.type == "conflict":
			response = "The accounts I considered cannot all be accurate. That does not identify anyone as evil." if best.agreement > 0 else "I have not established a conflict among the accounts I considered."
		elif best.weight < 0:
			response = "The evidence I considered weighs against this conclusion; my search was limited."
		elif not reasoning.support.is_empty() and reasoning.status == "provisional":
			response = "Your explanation fits what I considered. I am provisionally persuaded, but I have not checked another explanation." if stance == "accept" else "I found your explanation coherent, but I am not convinced by its support. I have not checked another explanation."
		elif reasoning.status == "rival_found":
			var alternative = Solver.describe_world(view, reasoning.rival, conclusion.at)
			response = "I checked another explanation and it also fits: %s evil; %s Marked. I cannot settle it from this alone." % [", ".join(PackedStringArray(alternative.evil.map(func(id): return Solver.person_name(view, id)))), Solver.person_name(view, alternative.marked)]
			if alternative.has("demon"):
				response += " In that explanation, %s is the Demon." % Solver.person_name(view, alternative.demon)
			if Movement.available(view) and stance == "accept":
				response = "I checked another explanation that also fits, but the movement and trace evidence makes your explanation more convincing. This is persuasion, not a proof that the alternative is impossible."
		elif reasoning.status == "entailed_in_attention":
			response = "The reports I considered force this if they are accurate."
		else:
			response = "I could not follow your explanation through within what I considered. That does not prove it wrong."
		if not reasoning.ignored.is_empty() or not reasoning.ignored_lines.is_empty():
			response += " I did not combine all the evidence in this case."
		if repeated:
			response = "That is the same case; I made the same checks. Repeating it adds no new reason."
	if not reasoning.private_checks.objections.is_empty():
		objection = " ".join(PackedStringArray(reasoning.private_checks.objections))
	if listener.get("reasoning_mode", "full") == "bounded" and combined.factors.any(func(label): return label.begins_with("Challenged home alibi")):
		response += " The linked observation challenges the home alibi; it does not by itself prove evil."
	if theory.mode == "persistence" and not repeated:
		response += " " + theory.note
	if not previous.is_empty() and not repeated and not changes.is_empty():
		response += " " + " ".join(PackedStringArray(changes))
	if not case_data.get("faults", {}).is_empty():
		response += " Your explanation needs impairment in the selected interval. That remains a hypothesis; readings outside it keep their own support."
	return {"listener": listener.id, "name": listener.name, "version": case_data.version, "stance": stance,
		"agreement": best.agreement, "quality": best.quality, "strength": strength, "path": best.cards,
		"path_kind": best.kind, "repeated": repeated, "response": response, "objection": objection, "changes": changes,
		"assessment": assessment, "shared_assessment": shared, "pattern": pattern, "support_model": combined, "legacy_path": legacy, "reasoning": reasoning, "theory": theory,
		"before_suspicion": before_suspicion, "after_suspicion": listener.suspicion[target], "suspicion_delta": delta,
		"before_trust": before_trust, "after_trust": listener.trust, "trust_delta": 0.0,
		"calculation": {"agreement": best.agreement, "premise_quality": best.quality, "player_factor": player_factor,
			"confidence_factor": confidence_factor, "coefficient": coefficient, "old_case_effect": old.get("applied", 0.0), "new_case_effect": applied,
			"venue": context.get("venue", "private"), "rule": "Replace this case's prior effect; never add it again. Trust is held fixed for controlled comparisons."}}

## The listener as CaseEval sees them: their suspicion and trust as priors, their own
## private readings folded into the starting view, and their reading reliability.
## Challenges the listener accepts (unlikely / misled / lying) scale the reliability of
## the challenged readings and sources.
static func _eval_listener(listener: Dictionary, own: Array, challenges: Dictionary = {}, attention: Dictionary = {}) -> Dictionary:
	var knowledge: Dictionary = listener.get("knowledge", {}).duplicate(true)
	if listener.get("reasoning_mode", "full") == "bounded" and knowledge.has("presented_reports") and not attention.is_empty():
		own = own.filter(func(id): return id in attention.considered)
	knowledge["own_reports"] = own
	return {"id": listener.get("id", "listener"), "suspicion": listener.get("suspicion", {}),
		"npc_trust": listener.get("npc_trust", {}), "knowledge": knowledge,
		"reading_reliability": listener.get("reading_reliability", {}),
		"challenge_reports": challenges.get("reports", {}), "challenge_sources": challenges.get("sources", {})}

## A bounded listener who looked for no rival explanation ("provisional"), or who
## checked that the attended reports force it ("entailed_in_attention"), takes a
## coherent case on trust: model v5's provisional conviction, scaled by how far they
## trust each source (Support.source_quality) and each reading. Supplied factors still
## move them.
## entailed: the listener checked that the attended reports force the conclusion, so
## with every reading credible they're fully convinced.
static func _provisional(view: Dictionary, conclusion: Dictionary, ev: Dictionary, cards: Array, listener: Dictionary, eval_listener: Dictionary, entailed: bool = false) -> float:
	var ratio: float = Support._direct_ratio(ev.factors.filter(func(f): return not f.get("joint_live", false)), conclusion)
	if cards.is_empty():
		return (2.0 if Movement.available(view) else 0.16) * (ratio / (1.0 + ratio) - 0.5)
	# Each source is credible with its quality; each of its readings, with that reading's
	# own reliability. Conviction is v5's for however many readings survive (two or
	# more: 0.76, one: 0.3), averaged over the chances.
	var by_source = {}
	for card in cards:
		if not by_source.has(card.source):
			by_source[card.source] = []
		by_source[card.source].append(minf(1.0, (1.0 - CaseEval.reading_error(view, card, eval_listener)) / (1.0 - CaseEval.DEFAULT_ERR)))
	var dist = PackedFloat64Array([1.0])  # chance of n credible readings
	for source in by_source:
		var quality: float = Support.source_quality(listener, source)
		var inner = PackedFloat64Array([1.0])  # given the source is credible
		for r in by_source[source]:
			var grown = PackedFloat64Array()
			grown.resize(inner.size() + 1)
			grown.fill(0.0)
			for n in range(inner.size()):
				grown[n] += inner[n] * (1.0 - r)
				grown[n + 1] += inner[n] * r
			inner = grown
		var next = PackedFloat64Array()
		next.resize(dist.size() + inner.size() - 1)
		next.fill(0.0)
		for a in range(dist.size()):
			next[a] += dist[a] * (1.0 - quality)
			for b in range(inner.size()):
				next[a + b] += dist[a] * quality * inner[b]
		dist = next
	var scale: float = 2.0 * ratio / (1.0 + ratio)
	var conviction = 0.0
	for n in range(1, dist.size()):
		conviction += dist[n] * (1.0 if entailed and n == cards.size() else minf(0.95, (0.76 if n >= 2 else 0.3) * scale))
	return conviction

## The cards a listener weighs for this case: their attention set (Insight limits it for
## bounded reasoners), without reports they know are false and without their own private
## readings, which are part of their starting view instead.
static func _eval_case(view: Dictionary, case_data: Dictionary, attention: Dictionary, listener: Dictionary) -> Dictionary:
	var scoped: Dictionary = attention.case.duplicate(true)
	var own: Array = listener.get("knowledge", {}).get("own_reports", [])
	scoped.cards = scoped.cards.filter(func(id): return id not in attention.private_checks.rejected and id not in own)
	scoped.conclusion = case_data.conclusion.duplicate(true)
	if case_data.conclusion.type in ["suspicious"]:
		scoped.conclusion.type = "evil"
	return scoped

## Headline judgment on the CaseEval engine. Same result shape as _assess_target_inner.
static func _assess_target_eval(view: Dictionary, case_data: Dictionary, listener: Dictionary, context: Dictionary = {}) -> Dictionary:
	var span = Profiler.open("headline")
	var original = case_data.duplicate(true)
	case_data = Graph.compile_case(view, case_data)
	var conclusion: Dictionary = case_data.conclusion
	var target: String = conclusion.target
	var before_suspicion: float = listener.suspicion.get(target, 0.1)
	var before_trust: float = listener.trust
	var old: Dictionary = listener.ledger.get(case_data.id, {})
	if not old.is_empty():
		listener.suspicion[old.target] = snappedf(listener.suspicion.get(old.target, 0.1) - old.applied, 0.000000001)
	var previous: Dictionary = listener.case_memory.get(case_data.id, {})
	var signature = _signature(view, case_data)
	var input_key = JSON.stringify(_canonical([signature, listener.get("reasoning_mode", "full"), listener.get("insight", 0.5),
		listener.get("confidence_mode", "scale"), listener.confidence, listener.trust, listener.npc_trust,
		listener.suspicion, listener.reading_reliability, listener.knowledge]))
	var repeated: bool = previous.get("input_key", "") == input_key
	var knowledge: Dictionary = listener.knowledge
	var full_case = case_data.duplicate(true)
	for id in knowledge.get("own_reports", []):
		if id not in full_case.cards:
			full_case.cards.append(id)
	# Live belief/attention/action never read the exhaustive minimal-conflict list.
	# Keep it available to the explicit diagnostic UI, out of every receipt refresh.
	var assessment = Solver.analyze(view, full_case, knowledge, false)
	var shared = Solver.analyze(view, case_data, {}, false)
	var reasoning = Support.explain(view, original, listener)
	var bounded: bool = listener.get("reasoning_mode", "full") == "bounded"
	if bounded:
		var considered_facts: Array = reasoning.considered.map(func(id): return Solver.semantic_key(Solver.card_by_id(view, id)))
		considered_facts.sort()
		input_key += JSON.stringify([considered_facts, reasoning.connections])
		repeated = previous.get("input_key", "") == input_key
	var theory = Theory.prepare(view, case_data, listener, reasoning, input_key)
	var attention = Reasoner.prepare(view, original, listener)
	var scoped = _eval_case(view, case_data, attention, listener)
	var challenges = Support.ReadingChallenges.assess(view, attention.case, listener, Solver.selected(view, attention.case).filter(func(c): return c.id not in attention.private_checks.rejected))
	var eval_listener = _eval_listener(listener, knowledge.get("own_reports", []), challenges, attention)
	eval_listener["theory_bias"] = theory.bias
	var ev = _case_eval(view, scoped, eval_listener)
	var agreement: float = clampf(CaseEval.shift(ev.prior, ev.posterior), -1.0, 1.0)
	var contributing: Array = Solver.selected(view, scoped).filter(func(c): return Solver.tabled(c))
	if bounded and reasoning.status in ["provisional", "entailed_in_attention"] and ev.consistent and conclusion.type in ["evil", "good", "suspicious", "marked", "demon", "minion"]:
		agreement = clampf(_provisional(view, conclusion, ev, contributing, listener, eval_listener, reasoning.status == "entailed_in_attention"), -1.0, 1.0)
	var pattern = _pattern(view, case_data, listener)
	if bounded:
		# Outings count only if the listener attended to them.
		var attended_case: Dictionary = attention.case.duplicate(true)
		attended_case.cards = attended_case.cards.filter(func(id): return id not in reasoning.private_checks.rejected)
		pattern = _pattern(view, attended_case, listener)
	var paths: Array = scoped.cards.filter(func(id): return Solver.tabled(Solver.card_by_id(view, id)))
	var best = {"weight": agreement, "agreement": agreement, "quality": 1.0, "cards": paths, "kind": "reading"}
	if conclusion.type == "suspicious" and pattern.weight > best.weight:
		best = pattern
	# The player's chosen answers and positions are a theory: if the conclusion follows
	# only once they're granted, say so. They never add evidence.
	if best.kind == "reading" and best.agreement < 0.6 and conclusion.type not in ["conflict", "suspicious"]:
		var positions: Dictionary = case_data.positions.duplicate(true)
		positions.erase(target)
		var answers: Dictionary = case_data.answers.duplicate(true)
		if not positions.is_empty() or not answers.is_empty() or not case_data.get("temporal_positions", []).is_empty():
			var chosen = Solver.solve(view, Solver.selected(view, full_case).filter(func(c): return c.id not in reasoning.private_checks.rejected), positions, answers, knowledge, conclusion.at, case_data.get("temporal_positions", []))
			if not chosen.worlds.is_empty() and chosen.worlds.all(func(w): return Solver.holds(view, w, conclusion)):
				best.kind = "conditional"
	var conflict = assessment.status == "conflicting" or not ev.consistent
	if conclusion.type == "conflict":
		var detected: bool = (reasoning.status == "inconsistent" or not reasoning.private_checks.rejected.is_empty()) if bounded else not ev.consistent
		var value = 1.0 if detected else 0.0
		best = {"weight": value, "agreement": value, "quality": 1.0, "cards": case_data.cards, "kind": "conflict"}
	if reasoning.private_checks.fatal or ev.impossible:
		best.weight = 0.0
		best.agreement = 0.0
	var player_factor: float = 0.5 + listener.trust
	var confidence_factor: float = 1.0 if listener.get("confidence_mode", "scale") == "persistence" else 1.4 - 0.8 * listener.confidence
	var strength: float = best.weight * player_factor * confidence_factor
	var coefficient: float = {"evil": 0.45, "demon": 0.45, "minion": 0.45, "good": -0.4, "marked": -0.1, "suspicious": 0.3, "conflict": 0.0}.get(conclusion.type, 0.0)
	var effect: float = strength * coefficient
	var baseline: float = listener.suspicion.get(target, 0.1)
	listener.suspicion[target] = snappedf(clampf(baseline + effect, 0.0, 1.5), 0.000000001)
	var applied: float = snappedf(listener.suspicion[target] - baseline, 0.000000001)
	listener.ledger[case_data.id] = {"target": target, "applied": applied, "cards": best.cards.duplicate(), "version": case_data.version,
		"conclusion": conclusion.duplicate(true), "signature": signature, "report_acceptance": ev.acceptance.duplicate(true)}
	var changes = _changes(view, previous.get("case", {}), case_data)
	listener.case_memory[case_data.id] = {"signature": signature, "input_key": input_key, "version": case_data.version, "cards": case_data.cards.duplicate(), "strength": strength, "case": case_data.duplicate(true)}
	var objection = _objection(view, case_data, listener, shared, assessment, best, pattern)
	var stance = "accept" if best.agreement >= 0.6 else ("partial" if best.agreement >= 0.35 else "reject")
	var response = _eval_response(view, case_data, best, stance, ev, conflict, repeated)
	if bounded:
		# What a bounded listener says reflects their own search, never the referee's.
		objection = _bounded_objection(reasoning)
		if reasoning.private_checks.fatal:
			response = reasoning.private_checks.objections[0]
		elif conclusion.type == "conflict":
			response = "The accounts I considered cannot all be accurate. That does not identify anyone as evil." if best.agreement > 0 else "I have not established a conflict among the accounts I considered."
		elif best.weight < 0:
			response = "The evidence I considered weighs against this conclusion; my search was limited."
		elif not reasoning.support.is_empty() and reasoning.status == "provisional":
			response = "Your explanation fits what I considered. I am provisionally persuaded, but I have not checked another explanation." if stance == "accept" else "I found your explanation coherent, but I am not convinced by its support. I have not checked another explanation."
		elif reasoning.status == "rival_found":
			var alternative = Solver.describe_world(view, reasoning.rival, conclusion.at)
			response = "I checked another explanation and it also fits: %s evil; %s Marked." % [", ".join(PackedStringArray(alternative.evil.map(func(id): return Solver.person_name(view, id)))), Solver.person_name(view, alternative.marked)]
			if alternative.has("demon"):
				response += " In that explanation, %s is the Demon." % Solver.person_name(view, alternative.demon)
			if stance == "accept":
				response += " Even so, the evidence favors your explanation." if ev.movement.is_empty() else " Even so, the movement and trace evidence makes your explanation more convincing. This is persuasion, not a proof that the alternative is impossible."
		elif reasoning.status == "entailed_in_attention":
			response = "The reports I considered force this if they are accurate."
		elif reasoning.status != "provisional":
			response = "I could not follow your explanation through within what I considered. That does not prove it wrong."
		if not reasoning.ignored.is_empty() or not reasoning.ignored_lines.is_empty():
			response += " I did not combine all the evidence in this case."
		if ev.movement.any(func(label): return label.begins_with("Challenged home alibi")):
			response += " The linked observation challenges the home alibi; it does not by itself prove evil."
		if repeated:
			response = "That is the same case; I made the same checks. Repeating it adds no new reason."
	for note in ev.movement:
		if note.begins_with("Seen elsewhere:") or note.begins_with("Scattered sightings:"):
			response += " " + note
	if not reasoning.private_checks.objections.is_empty():
		objection = " ".join(PackedStringArray(reasoning.private_checks.objections))
	if theory.mode == "persistence" and not repeated:
		response += " " + theory.note
	if not previous.is_empty() and not repeated and not changes.is_empty():
		response += " " + " ".join(PackedStringArray(changes))
	if not case_data.get("faults", {}).is_empty():
		response += " Your explanation needs impairment in the selected interval. That remains a hypothesis; readings outside it keep their own support."
	var support_model = {"agreement": agreement, "evidence_agreement": agreement, "forced_mass": ev.posterior if ev.posterior >= 0.999 else 0.0,
		"conditional_mass": 0.0, "paths": [paths], "report_acceptance": ev.acceptance, "states": 1, "factors": ev.movement, "atoms": [], "credence": ev.credence,
		"quoted_cards": scoped.get("quoted_cards", []).duplicate(), "faults": [], "movement_comparison": [], "prior": ev.prior, "posterior": ev.posterior,
		"category": ev.category, "rivals": ev.rivals, "theory_fit": ev.theory_fit,
		"model": "CaseEval: likelihood-weighted worlds (docs/CASE_REASONING.md)."}
	Profiler.close(span)
	return {"listener": listener.id, "name": listener.name, "version": case_data.version, "stance": stance,
		"agreement": best.agreement, "quality": best.quality, "strength": strength, "path": best.cards,
		"path_kind": best.kind, "repeated": repeated, "response": response, "objection": objection, "changes": changes,
		"assessment": assessment, "shared_assessment": shared, "pattern": pattern, "support_model": support_model, "legacy_path": {"weight": 0.0, "omitted": true},
		"reasoning": reasoning, "theory": theory, "case_eval": ev,
		"before_suspicion": before_suspicion, "after_suspicion": listener.suspicion[target], "suspicion_delta": listener.suspicion[target] - before_suspicion,
		"before_trust": before_trust, "after_trust": listener.trust, "trust_delta": 0.0,
		"calculation": {"agreement": best.agreement, "premise_quality": best.quality, "player_factor": player_factor,
			"confidence_factor": confidence_factor, "coefficient": coefficient, "old_case_effect": old.get("applied", 0.0), "new_case_effect": applied,
			"venue": context.get("venue", "private"), "prior": ev.prior, "posterior": ev.posterior,
			"rule": "Replace this case's prior effect; never add it again. Trust is held fixed for controlled comparisons."}}

## What the listener says about the headline, from how far the case moved them.
static func _eval_response(view: Dictionary, case_data: Dictionary, best: Dictionary, stance: String, ev: Dictionary, conflict: bool, repeated: bool) -> String:
	var conclusion: Dictionary = case_data.conclusion
	if repeated:
		return "That is the same case. Repeating it adds no new reason."
	if conclusion.type == "conflict":
		return "Those accounts cannot all be accurate. I have reopened which report to believe; this does not establish anyone's alignment." if best.agreement > 0 else "I don't see a conflict among the reports you selected."
	if best.kind == "pattern" and best.weight > 0:
		return "That gives me a reason to be suspicious, though it does not establish evil."
	var response = ""
	if best.kind == "conditional":
		response = "That follows if I grant your extra choices. You still need to establish those choices."
	elif ev.impossible:
		response = "Your selected explanation does not fit what I know. The underlying reports can still change my assessments."
	elif conflict:
		response = "Those reports can't all be right together, so I'm weighing which of them to doubt."
	elif stance == "accept":
		response = "Taken together, those reports convince me." if ev.posterior >= 0.9 else "Those reports convince me, though another explanation is still possible."
	elif stance == "partial":
		response = "That moves me, but the reports also allow another explanation."
	elif best.agreement <= -0.2:
		response = "The supplied evidence weighs against that conclusion."
	elif best.agreement > 0.05:
		response = "I can see that possibility, but it doesn't settle much for me."
	else:
		response = "Those reports don't change my view of that."
	if stance != "accept" and not ev.rivals.is_empty():
		var rival: Dictionary = ev.rivals[0]
		if rival.shift > 0.2 and rival.posterior > 0.5:
			response += " If anything, it points me toward %s." % Solver.person_name(view, rival.id)
	if ev.movement.any(func(label): return label.begins_with("Movement / trace")):
		response += " The routes, observation coverage and traces change which explanation I favor. Lawful activity and eligible passers remain alternatives; this is circumstantial evidence."
	elif ev.movement.any(func(label): return label.begins_with("Challenged home alibi")):
		response += " If the linked observation is accurate, the home alibi is false. That is circumstantial evidence, not proof of deliberate deception or evil."
	elif not ev.movement.is_empty():
		response += " The supplied test factor changes which remaining explanation I favor; it does not prove it."
	return response

## Effects on everyone from one CaseEval pass over the same attended evidence, plus the
## intermediate claims. Same result shape as Consequences.assess.
## headline is {} for a belief refresh: no person is special, and refresh skips the
## intermediate claims and link diagnostics (they only explain a presented case).
static func _consequences_eval(view: Dictionary, value: Dictionary, baseline: Dictionary, headline: Dictionary, common_key: String, only: Array = [], refresh: bool = false) -> Dictionary:
	var span = Profiler.open("consequences")
	var attention = Reasoner.prepare(view, value, baseline)
	var compiled = Graph.compile_case(view, value)
	var scoped = _eval_case(view, compiled, attention, baseline)
	var attended: Array = Solver.selected(view, attention.case).filter(func(c): return c.id not in attention.private_checks.rejected)
	var challenged = Support.ReadingChallenges.assess(view, attention.case, baseline, attended)
	var eval_listener = _eval_listener(baseline, baseline.knowledge.get("own_reports", []), challenged, attention)
	var ev: Dictionary = _case_eval(view, scoped, eval_listener)
	var protection = Protection.assess(view, attended, baseline)
	# The headline's own fatal check (say, accusing the listener) doesn't void what the
	# evidence says about everyone else: each person gets their own direct check.
	var invalid: bool = not attention.case.get("graph_errors", []).is_empty() or not Solver.validate_answers(view, attention.case).is_empty()
	var persistence: bool = baseline.get("confidence_mode", "scale") == "persistence"
	var confidence: float = 1.0 if persistence else 1.4 - 0.8 * baseline.confidence
	var theories: Dictionary = baseline.get("theories", {}).duplicate(true)
	var effects = {}
	for person in view.people:
		var id: String = person.id
		if not only.is_empty() and id not in only:
			continue
		var is_headline: bool = not headline.is_empty() and id == value.conclusion.target
		var theory: Dictionary = headline.theory if is_headline else {}
		# A refresh has no headline search; its effects all report the refresh itself.
		var status: String = headline.reasoning.status if not headline.is_empty() else "belief_refresh"
		var person_ev: Dictionary = ev
		if persistence:
			# A committed listener keeps a working theory per person and reads the
			# evidence through it (Theory bias), as the headline does.
			var evil_case: Dictionary = attention.case.duplicate(true)
			evil_case.conclusion = {"type": "evil", "target": id, "at": value.conclusion.at}
			# The explanation search only feeds Theory.prepare. Skip it when its result
			# would be unused: the headline's own theory, or a decision Theory already
			# has (unchanged inputs, or the fixed commitments a scoped reply inherits).
			if not is_headline or value.conclusion.type not in ["evil", "good", "suspicious"]:
				var known: Dictionary = Theory.cached(view, evil_case, baseline, common_key + id)
				if not known.is_empty():
					theory = known
				else:
					var clone = baseline.duplicate(true)
					var reasoning = Support.explain(view, evil_case, clone, Consequences._prepared(view, attention, evil_case, baseline))
					theory = Theory.prepare(view, evil_case, clone, reasoning, common_key + id)
					for key in clone.get("theories", {}):
						if key.begins_with(id + "|"):
							theories[key] = clone.theories[key]
			if not theory.get("bias", {}).is_empty():
				var biased = eval_listener.duplicate(true)
				biased["theory_bias"] = theory.bias
				person_ev = _case_eval(view, scoped, biased)
		var odds: Dictionary = person_ev.people.get(id, {"prior": 0.0, "posterior": 0.0})
		var s: float = CaseEval.shift(odds.prior, odds.posterior) if not ev.impossible else 0.0
		var alignment: float = 0.45 * s
		var circumstantial = 0.0
		if not Movement.available(view):
			var pattern_case = attention.case.duplicate(true)
			pattern_case.conclusion = {"type": "suspicious", "target": id, "at": value.conclusion.at}
			pattern_case.cards = pattern_case.cards.filter(func(c): return c not in attention.private_checks.rejected)
			circumstantial = 0.3 * _pattern(view, pattern_case, baseline).weight
		var net_weight: float = circumstantial if circumstantial > absf(alignment) else alignment
		var protection_signal: float = -protection.get(id, {}).get("weight", 0.0)
		if absf(protection_signal) > absf(net_weight):
			net_weight = protection_signal
		for challenge in challenged.results:
			if challenge.mode == "lying" and challenge.source == id and challenge.agreement >= 0.6:
				net_weight = maxf(net_weight, 0.12 * challenge.agreement)
		var own_check: Dictionary = attention.case.duplicate(true)
		own_check.conclusion = {"type": "evil", "target": id, "at": value.conclusion.at}
		if invalid or Reasoner.direct_checks(view, own_check, baseline.knowledge).fatal or TimeModel.known_good(view, baseline.knowledge, id, value.conclusion.at) or value.conclusion.type == "conflict":
			net_weight = 0.0
		effects[id] = {"requested": net_weight * (0.5 + baseline.trust) * confidence, "alignment_signal": alignment, "pattern_signal": circumstantial,
			"route": "protection" if protection_signal != 0.0 and net_weight == protection_signal else ("circumstantial" if circumstantial > absf(alignment) else "alignment"),
			"protection_signal": protection_signal, "evil_agreement": odds.posterior, "good_agreement": 1.0 - odds.posterior,
			"evil_change": s, "good_change": -s, "prior": odds.prior, "posterior": odds.posterior,
			"cards": attention.considered.duplicate(), "theory": theory, "reasoning": status}
	var claims: Array = []
	var by_id = {}
	var arguments: Array = [] if refresh else Consequences._arguments(view, value)
	for argument in arguments:
		var claim_case = Consequences._scope(view, argument.case, attention.considered)
		claims.append(_eval_claim(view, argument, claim_case, attention, baseline, eval_listener))
		by_id[argument.id] = claims[-1]
	# Links: how the parent claim fares without the evidence only this branch brings.
	for argument in arguments:
		var child: Dictionary = by_id[argument.id]
		for parent_id in argument.parents:
			if not by_id.has(parent_id):
				continue
			var parent: Dictionary = by_id[parent_id]
			var parent_argument: Dictionary = arguments.filter(func(a): return a.id == parent_id)[0]
			var retained: Array = argument.get("without", {}).get(parent_id, [])
			if not argument.has("without"):
				for input in parent_argument.get("inputs", []):
					if input.id != argument.id:
						retained.append_array(input.cards)
			var exclusive: Array = child.cards.filter(func(id): return id not in retained)
			var without = Consequences._scope(view, parent_argument.case, parent.cards.filter(func(id): return id not in exclusive))
			var reduced = _eval_claim(view, parent_argument, without, attention, baseline, eval_listener)
			var contribution: float = parent.agreement - reduced.agreement
			var relation = "shared" if exclusive.is_empty() else ("strengthens" if contribution > 0.0000001 else ("weakens" if contribution < -0.0000001 else "no_preference"))
			var remaining: bool = not parent.reasoning.get("rival", {}).is_empty()
			var note = {"shared": "These original sources are already included through another connection.",
				"strengthens": "This branch's evidence strengthens the parent claim.",
				"weakens": "This branch's evidence weighs against the parent claim.",
				"no_preference": "This branch does not add a preference for the parent claim."}[relation]
			if child.stance == "reject":
				note += " I do not accept the intermediate claim; its label supplies no extra evidence."
			if remaining:
				note += " An alternative to the parent claim still fits what I checked."
			child.links.append({"parent": parent_id, "parent_label": parent.label, "relation": relation,
				"evidence_difference": contribution, "exclusive_cards": exclusive, "alternative_remains": remaining, "note": note})
	Profiler.close(span)
	return {"effects": effects, "claims": claims, "theories": theories, "considered": attention.considered,
		"reading_challenges": challenged.results,
		"reasoning": {"status": "belief_refresh", "impossible": ev.impossible, "profile": attention.profile, "considered": attention.considered.duplicate(),
			"ignored": attention.ignored.duplicate(), "ignored_lines": attention.ignored_lines.duplicate(), "private_checks": attention.private_checks.duplicate(true)},
		"rule": "One CaseEval pass over the attended evidence gives every person's change; intermediate claims add no evidence."}

## One claim (the headline or an intermediate) judged on the attended evidence it cites.
static func _eval_claim(view: Dictionary, argument: Dictionary, claim_case: Dictionary, attention: Dictionary, baseline: Dictionary, eval_listener: Dictionary) -> Dictionary:
	var checks = Reasoner.direct_checks(view, claim_case, baseline.knowledge)
	var scoped: Dictionary = claim_case.duplicate(true)
	var own: Array = baseline.knowledge.get("own_reports", [])
	scoped.cards = scoped.cards.filter(func(id): return id not in checks.rejected and id not in own)
	if scoped.conclusion.type == "suspicious":
		scoped.conclusion.type = "evil"
	var ev = _case_eval(view, scoped, eval_listener)
	# The logical layer (what follows if the attended reports are sound) is the
	# reasoner's; how much the listener believes it is CaseEval's.
	var thought = Support.explain(view, claim_case, baseline, Consequences._prepared(view, attention, claim_case, baseline))
	var status: String = thought.status
	var agreement: float = clampf(CaseEval.shift(ev.prior, ev.posterior), -1.0, 1.0)
	var pattern: Dictionary = {"weight": 0.0}
	if claim_case.conclusion.type == "suspicious":
		var pattern_case = claim_case.duplicate(true)
		pattern_case.cards = pattern_case.cards.filter(func(id): return id not in checks.rejected)
		pattern = _pattern(view, pattern_case, baseline)
		agreement = maxf(agreement, pattern.agreement if pattern.weight > 0 else 0.0)
	if claim_case.conclusion.type == "conflict":
		agreement = 1.0 if status == "inconsistent" or not ev.consistent or not checks.rejected.is_empty() else 0.0
	if checks.fatal or ev.impossible:
		agreement = 0.0
	var note = "Another explanation remains possible."
	if claim_case.cards.is_empty():
		note = "No attended evidence was offered for this claim."
	elif checks.fatal:
		note = " ".join(PackedStringArray(checks.objections))
	elif claim_case.conclusion.type == "suspicious" and pattern.weight > 0:
		note = "The observations give a circumstantial reason for suspicion; they do not establish evil."
	elif claim_case.conclusion.type == "suspicious" and not ev.movement.is_empty() and agreement > 0:
		note = "The joint movement and trace comparison gives a circumstantial reason for suspicion; it does not establish evil."
	elif claim_case.conclusion.type == "conflict":
		note = "The considered accounts conflict; that does not identify deliberate deception or evil." if agreement > 0 else "I have not established a conflict in the considered accounts."
	elif status == "inconsistent" or not ev.consistent:
		note = "The premises I checked cannot all hold together."
	elif status == "entailed_in_attention":
		note = "This follows from the reports I granted, if those reports are sound."
	elif agreement >= 0.6:
		note = "The evidence favors this, though another explanation is still possible."
	elif status == "provisional":
		note = "This fits the evidence I checked; I have not checked another explanation."
	elif agreement <= -0.2:
		note = "The evidence weighs against this."
	var scoped_protection = Protection.assess(view, Solver.selected(view, claim_case).filter(func(c): return c.id not in checks.rejected), baseline)
	if claim_case.conclusion.type == "good" and scoped_protection.has(claim_case.conclusion.target) and not checks.fatal:
		agreement = maxf(agreement, minf(0.55, 0.35 + scoped_protection[claim_case.conclusion.target].weight))
		note = "A quiet night fits this protection claim, but it does not establish that an attack targeted their protection. Other explanations remain."
	return {"id": argument.id, "label": argument.label, "conclusion": claim_case.conclusion.duplicate(true),
		"parents": argument.parents.duplicate(), "cards": claim_case.cards.duplicate(),
		"ignored": Graph.compile_case(view, argument.case).cards.filter(func(id): return id not in claim_case.cards),
		"agreement": agreement, "stance": Consequences._stance(agreement), "status": status,
		"conditional_mass": 0.0, "forced_mass": 1.0 if status == "entailed_in_attention" and ev.consistent else 0.0,
		"note": note, "links": [], "support_model": {"agreement": agreement, "prior": ev.prior, "posterior": ev.posterior, "category": ev.category},
		"reasoning": thought}

static func _bounded_objection(reasoning: Dictionary) -> String:
	# Spoken objections use only this search, never the referee's alternative.
	var result: String = {"provisional": "I have not established that this is the only explanation.",
		"rival_found": "Both explanations fit the reports I granted for this comparison.",
		"entailed_in_attention": "This follows within the reports I considered, if they are accurate.",
		"inconsistent": "The accounts I checked cannot all be accurate together.",
		"counterexample_found": "I checked an alternative, but did not establish your proposed explanation.",
		"unresolved": "I could not make all the pieces fit, but I may be missing something."}.get(reasoning.status, "")
	if not reasoning.doubted.is_empty():
		result += " I still doubt these reports: " + ", ".join(PackedStringArray(reasoning.doubted)) + "."
	return result

static func _quality(view: Dictionary, ids: Array, listener: Dictionary) -> float:
	var sources: Array = []
	var quality = 1.0
	for id in ids:
		var card = Solver.card_by_id(view, id)
		if card.is_empty():
			continue
		var source: String = card.source
		if source == listener.id or source in sources:
			continue
		sources.append(source)
		if source == "player":
			quality *= 0.9
		else:
			quality *= clampf(0.2 + 0.65 * listener.npc_trust.get(source, 0.5) - 0.35 * listener.suspicion.get(source, 0.1), 0.05, 0.95)
	return quality

static func _possibility_agreement(view: Dictionary, worlds: Array, conclusion: Dictionary, listener: Dictionary) -> float:
	# Compare the listener's strongest supporting and rival explanations, as Argue
	# does. This is an uncalibrated disposition score, not a fraction of worlds.
	var supporting = 0.0
	var rival = 0.0
	for world in worlds:
		var score = 1.0
		for i in range(view.people.size()):
			var id: String = view.people[i].id
			var suspicion: float = listener.suspicion.get(id, 0.1)
			var trust: float = listener.npc_trust.get(id, 0.5)
			if world.evil & (1 << i):
				score *= maxf(0.15 + suspicion * 1.2 - (trust - 0.5) * 0.5, 0.02)
			else:
				score *= maxf(0.3 + (0.6 - suspicion) * 0.6 + (trust - 0.5) * 0.4, 0.05)
		if Solver.holds(view, world, conclusion):
			supporting = maxf(supporting, score)
		else:
			rival = maxf(rival, score)
	return 0.02 + 0.16 * supporting / maxf(supporting + rival, 0.000000001)

static func _pattern(view: Dictionary, case_data: Dictionary, listener: Dictionary) -> Dictionary:
	if Movement.available(view):
		return {"weight": 0.0, "agreement": 0.0, "quality": 0.0, "cards": [], "kind": "joint_movement", "incidents": 0, "alibi_conflicts": []}
	var target: String = case_data.conclusion.target
	var incidents = {}
	var contradictions: Array = []
	var cards = Solver.selected(view, case_data)
	for card in cards:
		var out: bool = (card.kind == "sighting" and target in card.get("subjects", [card.subject])) or (card.kind == "report" and card.get("role", "") == "nightwatch" and target in card.data.seen)
		if not out:
			continue
		var key = "%s:%d" % [target, card.night]
		var q = _quality(view, [card.id], listener)
		if not incidents.has(key) or q > incidents[key].quality:
			incidents[key] = {"quality": q, "card": card.id, "night": card.night}
	var weight = 0.0
	var path: Array = []
	for key in incidents:
		var incident: Dictionary = incidents[key]
		var explanation: String = case_data.reasons.get(key, "unexplained")
		var explained = 0.0
		if explanation == "gift":
			var claim = Graph.valid_citation(view, case_data, target, incident.night)
			if not claim.is_empty():
				explained = 0.8 * _quality(view, [claim.id], listener)
				path.append(claim.id)
		# Merely selecting 'gift' without supporting testimony is not an explanation.
		weight += 0.3 * incident.quality * (1.0 - explained)
		path.append(incident.card)
		for card in cards:
			if card.kind == "alibi" and card.subject == target and card.night == incident.night:
				weight += 0.2 * minf(incident.quality, _quality(view, [card.id], listener))
				contradictions.append(card.id)
				path.append(card.id)
	return {"weight": minf(weight, 0.8), "agreement": 0.5 if weight > 0 else 0.0, "quality": minf(weight / 0.5, 1.0),
		"cards": path, "kind": "pattern", "incidents": incidents.size(), "alibi_conflicts": contradictions}

static func _objection(view: Dictionary, case_data: Dictionary, listener: Dictionary, shared: Dictionary, assessment: Dictionary, best: Dictionary, pattern: Dictionary) -> String:
	if best.kind == "pattern" and best.weight > 0:
		if not pattern.alibi_conflicts.is_empty():
			return "The outing conflicts with the home alibi. Someone's account is wrong; that adds suspicion but does not identify evil."
		return "These are %d distinct outings. A credible night gift can explain an outing; seeing someone twice on one night is one incident." % pattern.incidents
	if assessment.status == "conflicting":
		if not assessment.errors.is_empty():
			return assessment.errors[0]
		if shared.status != "conflicting":
			return "That conflicts with my own gift or my knowledge that I am good."
		return "The asserted reports, answers or home alibis cannot all hold together. Remove or challenge a premise."
	if best.kind == "conditional":
		return "Your selected people narrow the world; choosing an answer does not establish it."
	if best.quality < 0.45 and not best.cards.is_empty():
		var weakest = ""
		var lowest = 2.0
		for id in best.cards:
			var q = _quality(view, [id], listener)
			if q < lowest:
				lowest = q
				weakest = id
		return "I am least convinced by %s's report (%s); its source and health assumptions still matter." % [Solver.person_name(view, Solver.card_by_id(view, weakest).source), weakest]
	if not assessment.alternative.is_empty() and assessment.entailment == "possible":
		var a: Dictionary = assessment.alternative
		if a.has("demon"):
			return "An alternative still fits with %s as the Demon; the other evil residents could be minions." % Solver.person_name(view, a.demon)
		return "An alternative still fits: %s evil; %s Marked." % [", ".join(PackedStringArray(a.evil.map(func(id): return Solver.person_name(view, id)))), Solver.person_name(view, a.marked)]
	if assessment.entailment == "unsupported":
		return "The accepted reports exclude that conclusion."
	return "This depends on the named sources being honest, holding their claimed gifts and being healthy at the reported times."

static func _signature(view: Dictionary, case_data: Dictionary) -> String:

		var positive: Array = case_data.answers.get(card.id, {}).get("positive", []).duplicate()
		if not positive.is_empty():

			answers[Solver.semantic_key(card)] = positive

		citations[key] = Solver.semantic_key(Solver.card_by_id(view, case_data.citations[key]))
	var scope: Array = Graph.roots(case_data, case_data.get("conflict_scope", [])).map(func(id): return Solver.semantic_key(Solver.card_by_id(view, id)))
	scope.sort()

		if not is_zero_approx(float(factor.log2_ratio)):
			var routes: Array = []
			for route in factor.routes:
				var facts: Array = route.map(func(id): return Solver.semantic_key(Solver.card_by_id(view, id)))
				facts.sort()
				routes.append(JSON.stringify(facts))
			routes.sort()
			factors.append(JSON.stringify([factor.target, factor.log2_ratio, factor.events, routes]))
	factors.sort()
	var timed: Array = []
	for assignment in case_data.get("temporal_positions", []):
		for id in assignment.positions:
			var atom = [assignment.at, id, assignment.positions[id]]
			if atom not in timed:
				timed.append(atom)
	timed.sort_custom(func(a, b): return JSON.stringify(a) < JSON.stringify(b))
	var signature = JSON.stringify(_canonical([keys, case_data.positions, answers, case_data.reasons, citations, scope, case_data.get("challenges", {}), factors, case_data.conclusion, case_data.get("graph_errors", []), case_data.get("faults", {}), timed]))
	return signature if case_data.get("reading_challenges", {}).is_empty() else signature + JSON.stringify(_canonical(case_data.reading_challenges))

static func _canonical(value):
	if value is Dictionary:
		var result = {}
		var keys = value.keys()
		keys.sort()
		for key in keys:
			result[key] = _canonical(value[key])
		return result
	if value is Array:
		return value.map(func(item): return _canonical(item))

static func _changes(view: Dictionary, previous: Dictionary, current: Dictionary) -> Array:
	var changes: Array = []
	if previous.is_empty():
		return changes
	for id in current.cards:
		if id not in previous.cards:
			var card = Solver.card_by_id(view, id)
			changes.append("You added %s's %s." % [Solver.person_name(view, card.source), card.get("role", card.kind)])
	for id in previous.cards:
		if id not in current.cards:
			var card = Solver.card_by_id(view, id)
			if card.is_empty():
				changes.append("You removed a previously cited report.")
			else:
				changes.append("You removed %s's %s." % [Solver.person_name(view, card.source), card.get("role", card.kind)])
	if previous.positions != current.positions:
		changes.append("You changed the assumed alignments or Marked identity.")
	if previous.answers != current.answers:
		changes.append("You changed the answers on your cards.")
	if previous.reasons != current.reasons:
		changes.append("You changed the explanations for the outings.")
	if previous.conclusion != current.conclusion:
		changes.append("You changed the conclusion.")
	return changes
